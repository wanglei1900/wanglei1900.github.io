class AuthenticationError extends GraphQLError {
  constructor(message) {
    super(message, {
      extensions: {
        code: 'UNAUTHENTICATED',

  }
}

  constructor(message, field) {
    super(message, {

}

3. Apollo Server 错误处理 ​

3.1 Apollo 错误类 ​

javascript
const { ApolloError, UserInputError, AuthenticationError, ForbiddenError } = require('apollo-server');

const resolvers = {
  Mutation: {
    createPost: (parent, { input }, context) => {
      // 认证错误
      if (!context.user) {
        throw new AuthenticationError('Authentication required');
      }
      
      // 权限错误
      if (!context.user.isAdmin) {
        throw new ForbiddenError('Insufficient permissions');
      }
      
      // 输入验证错误
      if (!input.title || input.title.length < 3) {
        throw new UserInputError('Invalid input', {
          invalidArgs: ['title'],
          details: 'Title must be at least 3 characters long'
        });
      }
      
      // 自定义业务错误
      if (isDuplicatePost(input.title)) {
        throw new ApolloError('Post with this title already exists', 'DUPLICATE_POST');
      }
      
      return createPost(input);
    }
  }
};

  context: ({ req }) => {
    // 全局错误处理上下文
    return {
      req,
      handleError: (error, code = 'INTERNAL_ERROR') => {
        throw new ApolloError(error.message, code);
      }
    };
  }
});

4. 高级错误处理模式 ​

4.1 错误包装器 ​

javascript
const withErrorHandling = (resolver) => {
  return async (parent, args, context, info) => {
    try {
      return await resolver(parent, args, context, info);
    } catch (error) {
      // 分类处理不同类型的错误
      if (error.name === 'ValidationError') {
        throw new UserInputError('Validation failed', {
          invalidArgs: Object.keys(error.errors),
          details: error.message
        });
      }
      
      if (error.code === 'SQLITE_CONSTRAINT') {
        throw new ApolloError('Database constraint violation', 'CONSTRAINT_VIOLATION');
      }
      
      // 记录未知错误
      console.error('Unhandled error in resolver:', {
        resolver: info.fieldName,
        error: error.message,
        stack: error.stack
      });
      
      throw new ApolloError('Internal server error', 'INTERNAL_SERVER_ERROR');
    }
  };
};

// 使用错误包装器
const resolvers = {
  Query: {
    user: withErrorHandling(async (parent, { id }) => {
      return await getUserById(id);
    })
  }
};

5. Schema 级别的错误处理 ​

5.1 联合类型错误模式 ​

graphql
# 定义可预期的错误类型
interface Error {
  message: String!
  code: String!
}

type UserNotFoundError implements Error {
  message: String!
  code: String!
  userId: ID!
}

type ValidationError implements Error {
  message: String!
  code: String!
  fields: [String!]!
}

# 成功和错误的联合类型
union CreateUserResult = User | UserNotFoundError | ValidationError

type Mutation {
  createUser(input: UserInput!): CreateUserResult!
}

# 查询示例
mutation {
  createUser(input: { email: "test@example.com" }) {
    ... on User {
      id
      email
    }
    ... on UserNotFoundError {
      message
      code
      userId
    }
    ... on ValidationError {
      message
      code
      fields
    }
  }
}

6.1 联合类型解析器 ​

const resolvers = {
  CreateUserResult: {
    __resolveType: (obj) => {
      if (obj.email) {
        return 'User';
      }
      if (obj.userId) {
        return 'UserNotFoundError';
      }
      if (obj.fields) {
        return 'ValidationError';
      }
      return null;
    }
  },
  
  Mutation: {
    createUser: async (_, { input }) => {
      // 验证输入
      const validationErrors = validateUserInput(input);
      if (validationErrors.length > 0) {
        return {
          __typename: 'ValidationError',
          message: 'Input validation failed',
          code: 'VALIDATION_ERROR',
          fields: validationErrors
        };
      }
      
      // 检查用户是否存在
      const existingUser = await findUserByEmail(input.email);
      if (existingUser) {
        return {
          __typename: 'ValidationError', 
          message: 'User already exists',
          code: 'DUPLICATE_USER',
          fields: ['email']
        };
      }
      
      try {
        const user = await createUser(input);
        return {
          __typename: 'User',
          ...user
        };
      } catch (error) {
        return {
          __typename: 'UserNotFoundError',
          message: 'Failed to create user',
          code: 'CREATION_FAILED',
          userId: 'unknown'
        };
      }
    }
  }
};

6.2 响应包装器解析器 ​

javascript
const resolvers = {
  Mutation: {
    createUser: async (_, { input }) => {
      const response = {
        code: '200',
        success: true,
        message: 'User created successfully',
        user: null,
        errors: []
      };
      
      try {
        // 输入验证
        const validationResult = validateUserInput(input);
        if (!validationResult.isValid) {
          response.code = '400';
          response.success = false;
          response.message = 'Validation failed';
          response.errors = validationResult.errors;
          return response;
        }
        
        // 创建用户
        const user = await createUser(input);
        response.user = user;
        return response;
        
      } catch (error) {
        response.code = '500';
        response.success = false;
        response.message = 'Internal server error';
        
        // 记录详细错误日志
        console.error('Create user error:', error);
        
        return response;
      }
    }
  }
};

7. 全局错误处理中间件 ​

7.1 Express 中间件集成 ​

javascript
const errorMiddleware = (app) => {
  // 捕获未处理的 Promise 拒绝
  process.on('unhandledRejection', (reason, promise) => {
    console.error('Unhandled Rejection at:', promise, 'reason:', reason);
  });
  
  process.on('uncaughtException', (error) => {
    console.error('Uncaught Exception:', error);
    process.exit(1);
  });
  
  // GraphQL 错误处理
  app.use('/graphql', (req, res, next) => {
    const originalSend = res.send;
    
    res.send = function(data) {
      try {
        const parsed = JSON.parse(data);
        
        if (parsed.errors) {
          // 记录 GraphQL 错误
          parsed.errors.forEach(error => {
            logGraphQLError(error, req);
          });
          
          // 生产环境过滤敏感信息
          if (process.env.NODE_ENV === 'production') {
            parsed.errors = parsed.errors.map(sanitizeError);
          }
        }
        
        originalSend.call(this, JSON.stringify(parsed));
      } catch (e) {
        originalSend.call(this, data);
      }
    };
    
    next();
  });
  
  // 404 处理
  app.use('*', (req, res) => {
    res.status(404).json({
      errors: [{
        message: 'Route not found',
        path: req.originalUrl
      }]
    });
  });
  
  // 全局错误处理
  app.use((error, req, res, next) => {
    console.error('Global error handler:', error);
    
    res.status(500).json({
      errors: [{
        message: process.env.NODE_ENV === 'development' 
          ? error.message 
          : 'Internal server error',
        code: 'INTERNAL_SERVER_ERROR'
      }]
    });
  });
};

const logGraphQLError = (error, req) => {
  const logEntry = {
    timestamp: new Date().toISOString(),
    message: error.message,
    path: error.path,
    code: error.extensions?.code,
    operationName: req.body?.operationName,
    userAgent: req.get('User-Agent'),
    ip: req.ip
  };
  
  console.error('GraphQL Error Log:', logEntry);
};

const sanitizeError = (error) => {
  const sanitized = { ...error };
  
  // 移除敏感信息
  if (sanitized.extensions?.exception) {
    delete sanitized.extensions.exception;
  }
  
  if (sanitized.stack) {
    delete sanitized.stack;
  }
  
  return sanitized;
};

8.1 Apollo Client 错误处理 ​

javascript
import { ApolloClient, InMemoryCache, from } from '@apollo/client';
import { onError } from '@apollo/client/link/error';

// 错误链接
const errorLink = onError(({ graphQLErrors, networkError, operation }) => {
  if (graphQLErrors) {
    graphQLErrors.forEach(({ message, locations, path, extensions }) => {
      console.error(
        `[GraphQL error]: Message: ${message}, Path: ${path}, Code: ${extensions?.code}`
      );
      
      // 处理特定错误码
      switch (extensions?.code) {
        case 'UNAUTHENTICATED':
          // 重定向到登录页
          window.location.href = '/login';
          break;
        case 'FORBIDDEN':
          // 显示权限不足提示
          showNotification('Insufficient permissions', 'error');
          break;
        case 'VALIDATION_ERROR':
          // 处理验证错误
          handleValidationErrors(extensions.invalidArgs);
          break;
        default:
          // 显示通用错误提示
          showNotification(message, 'error');
      }
    });
  }
  
  if (networkError) {
    console.error(`[Network error]: ${networkError}`);
    showNotification('Network error occurred', 'error');
  }
});

const client = new ApolloClient({
  link: from([errorLink, httpLink]),
  cache: new InMemoryCache(),
  defaultOptions: {
    watchQuery: {
      errorPolicy: 'all'
    },
    query: {
      errorPolicy: 'all'
    },
    mutate: {
      errorPolicy: 'all'
    }
  }
});

  };
};

10. 监控和日志 ​

10.1 错误监控集成 ​

javascript
const Sentry = require('@sentry/node');
const { Severity } = require('@sentry/node');

const initializeErrorMonitoring = () => {
  Sentry.init({
    dsn: process.env.SENTRY_DSN,
    environment: process.env.NODE_ENV,
    integrations: [
      new Sentry.Integrations.GraphQL()
    ]
  });
};

const errorMonitoringPlugin = {
  requestDidStart() {
    return {
      didEncounterErrors(ctx) {
        if (!ctx.operation) return;
        
        for (const err of ctx.errors) {
          // 只记录服务器错误，不记录客户端错误
          if (err.originalError && isServerError(err.originalError)) {
            Sentry.withScope(scope => {
              scope.setTag('kind', ctx.operation.operation);
              scope.setExtra('query', ctx.request.query);
              scope.setExtra('variables', ctx.request.variables);
              
              if (err.path) {
                scope.addBreadcrumb({
                  category: 'query-path',
                  message: err.path.join(' > '),
                  level: Severity.Error
                });
              }
              
              Sentry.captureException(err.originalError);
            });
          }
        }
      }
    };
  }
};

const isServerError = (error) => {
  const clientErrorCodes = [
    'VALIDATION_ERROR',
    'USER_INPUT_ERROR',
    'UNAUTHENTICATED',
    'FORBIDDEN'
  ];
  
  return !clientErrorCodes.includes(error.extensions?.code);
};
