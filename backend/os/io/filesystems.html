操作系统 I/O 管理 - 文件系统学习笔记 ​

1. 文件系统概述 ​

1.1 文件系统定义 ​

文件系统是操作系统中负责管理持久性数据存储的组件，它提供了数据组织、存储、检索和保护的机制。

1.2 文件系统的作用 ​

  • 数据持久化：将数据永久保存在存储设备上
  • 命名空间管理：提供层次化的文件组织方式
  • 访问控制：管理用户对文件的读写权限
  • 数据保护：防止数据损坏和丢失
  • 性能优化：提高数据存取效率

// 路径解析过程示例
int resolve_path(const char *path, struct inode **result) {
    if (path[0] == '/') {
        current_inode = root_inode;  // 从根目录开始
    } else {
        current_inode = cwd_inode;   // 从当前目录开始
    }
    
    char *component = strtok(path, "/");
    while (component != NULL) {
        // 在current_inode中查找component
        inode_number = lookup_in_directory(current_inode, component);
        if (inode_number == 0) return -ENOENT;
        
        current_inode = get_inode(inode_number);
        component = strtok(NULL, "/");
    }
    
    *result = current_inode;
    return 0;
}

4.4 多级索引示例 ​

c
// 计算文件需要的索引层级
int get_block_level(uint32_t file_size, uint32_t block_size) {
    uint32_t direct_blocks = 12;
    uint32_t pointers_per_block = block_size / sizeof(uint32_t);
    
    uint32_t direct_max = direct_blocks * block_size;
    uint32_t indirect_max = direct_max + pointers_per_block * block_size;
    uint32_t double_indirect_max = indirect_max + 
                                  pointers_per_block * pointers_per_block * block_size;
    
    if (file_size <= direct_max) return 0;          // 直接索引
    if (file_size <= indirect_max) return 1;        // 一级间接
    if (file_size <= double_indirect_max) return 2; // 二级间接
    return 3;                                       // 三级间接
}

6. 空闲空间管理 ​

6.1 位图（Bitmap） ​

c
// 使用位图管理空闲块
struct free_space_bitmap {
    uint8_t *bitmap;        // 位图数组
    uint32_t total_blocks;  // 总块数
    uint32_t free_blocks;   // 空闲块数
};

// 查找空闲块
int find_free_block(struct free_space_bitmap *fsb) {
    for (int i = 0; i < (fsb->total_blocks + 7) / 8; i++) {
        if (fsb->bitmap[i] != 0xFF) {  // 有空闲位
            for (int j = 0; j < 8; j++) {
                if (!(fsb->bitmap[i] & (1 << j))) {
                    return i * 8 + j;
                }
            }
        }
    }
    return -1;  // 没有空闲块
}

7. 文件系统操作实现 ​

7.1 文件打开 ​

c
struct file_descriptor {
    int fd;                 // 文件描述符
    struct inode *inode;    // 对应的inode
    off_t offset;           // 当前文件偏移
    int flags;              // 打开标志
    mode_t mode;            // 打开模式
    int reference_count;    // 引用计数
};

// 文件描述符表
struct fd_table {
    struct file_descriptor *fds[1024];  // 文件描述符数组
    int next_fd;                        // 下一个可用fd
};

int sys_open(const char *pathname, int flags, mode_t mode) {
    // 1. 路径解析，找到inode
    struct inode *inode = path_lookup(pathname);
    if (!inode) return -ENOENT;
    
    // 2. 权限检查
    if (!permission_check(inode, flags)) return -EACCES;
    
    // 3. 分配文件描述符
    int fd = allocate_fd();
    if (fd < 0) return -EMFILE;
    
    // 4. 创建文件描述符结构
    struct file_descriptor *file = kmalloc(sizeof(struct file_descriptor));
    file->fd = fd;
    file->inode = inode;
    file->offset = 0;
    file->flags = flags;
    file->reference_count = 1;
    
    // 5. 添加到文件描述符表
    current_process->fd_table->fds[fd] = file;
    
    return fd;
}

7.2 文件读取 ​

c
ssize_t sys_read(int fd, void *buf, size_t count) {
    // 1. 获取文件描述符
    struct file_descriptor *file = current_process->fd_table->fds[fd];
    if (!file) return -EBADF;
    
    // 2. 检查读取权限
    if (!(file->flags & O_RDONLY)) return -EACCES;
    
    // 3. 计算实际读取字节数
    size_t bytes_remaining = file->inode->size - file->offset;
    size_t bytes_to_read = count < bytes_remaining ? count : bytes_remaining;
    
    // 4. 读取数据
    ssize_t bytes_read = file_read_data(file->inode, file->offset, buf, bytes_to_read);
    
    // 5. 更新文件偏移
    if (bytes_read > 0) {
        file->offset += bytes_read;
    }
    
    return bytes_read;
}

7.3 文件写入 ​

c
ssize_t sys_write(int fd, const void *buf, size_t count) {
    struct file_descriptor *file = current_process->fd_table->fds[fd];
    if (!file) return -EBADF;
    
    if (!(file->flags & (O_WRONLY | O_RDWR))) return -EACCES;
    
    // 检查是否需要扩展文件
    if (file->offset + count > file->inode->size) {
        int ret = extend_file(file->inode, file->offset + count);
        if (ret < 0) return ret;
    }
    
    // 写入数据
    ssize_t bytes_written = file_write_data(file->inode, file->offset, buf, count);
    
    if (bytes_written > 0) {
        file->offset += bytes_written;
        file->inode->mtime = current_time();
    }
    
    return bytes_written;
}

8. 目录操作 ​

8.1 目录创建 ​

c
int sys_mkdir(const char *pathname, mode_t mode) {
    // 1. 解析父目录
    char *parent_path = get_parent_path(pathname);
    char *dir_name = get_last_component(pathname);
    
    struct inode *parent_inode = path_lookup(parent_path);
    if (!parent_inode) return -ENOENT;
    
    // 2. 检查是否已存在
    if (lookup_in_directory(parent_inode, dir_name) != 0) {
        return -EEXIST;
    }
    
    // 3. 分配inode和数据块
    ino_t new_inode = allocate_inode();
    if (new_inode == 0) return -ENOSPC;
    
    uint32_t data_block = allocate_block();
    if (data_block == 0) {
        free_inode(new_inode);
        return -ENOSPC;
    }
    
    // 4. 初始化目录inode
    struct inode *dir_inode = get_inode(new_inode);
    dir_inode->mode = S_IFDIR | (mode & 0777);
    dir_inode->size = 0;
    dir_inode->links_count = 2;  // . 和 ..
    dir_inode->blocks = 1;
    
    // 5. 创建目录内容
    create_directory_contents(dir_inode, data_block, parent_inode->inode_number);
    
    // 6. 添加到父目录
    add_directory_entry(parent_inode, dir_name, new_inode, FT_DIRECTORY);
    
    return 0;
}

8.2 目录遍历 ​

c
struct linux_dirent {
    unsigned long  d_ino;     // inode号
    unsigned long  d_off;     // 到下一个dirent的偏移
    unsigned short d_reclen;  // 当前dirent长度
    char           d_name[];  // 文件名
};

int sys_getdents(int fd, struct linux_dirent *dirp, unsigned int count) {
    struct file_descriptor *file = current_process->fd_table->fds[fd];
    if (!file) return -EBADF;
    
    struct inode *dir_inode = file->inode;
    if (!S_ISDIR(dir_inode->mode)) return -ENOTDIR;
    
    // 读取目录内容
    char buffer[BLOCK_SIZE];
    uint32_t total_read = 0;
    uint32_t dir_offset = file->offset;
    
    while (total_read < count && dir_offset < dir_inode->size) {
        // 读取目录块
        uint32_t block_offset = dir_offset % BLOCK_SIZE;
        uint32_t block_num = dir_offset / BLOCK_SIZE;
        
        read_block(get_data_block(dir_inode, block_num), buffer);
        
        // 处理目录项
        struct directory_entry *entry = (struct directory_entry*)(buffer + block_offset);
        
        // 填充linux_dirent结构
        struct linux_dirent *dent = (struct linux_dirent*)((char*)dirp + total_read);
        dent->d_ino = entry->inode_number;
        dent->d_off = dir_offset + entry->entry_length;
        dent->d_reclen = sizeof(struct linux_dirent) + entry->name_length + 1;
        memcpy(dent->d_name, entry->name, entry->name_length);
        dent->d_name[entry->name_length] = '\0';
        
        total_read += dent->d_reclen;
        dir_offset += entry->entry_length;
    }
    
    file->offset = dir_offset;
    return total_read;
}

9.2 缓存读写 ​

// 带缓存的读取
ssize_t cached_read(struct inode *inode, off_t offset, void *buf, size_t count) {
    size_t bytes_read = 0;
    
    while (bytes_read < count) {
        uint32_t page_index = (offset + bytes_read) / PAGE_SIZE;
        uint32_t page_offset = (offset + bytes_read) % PAGE_SIZE;
        size_t bytes_in_page = MIN(count - bytes_read, PAGE_SIZE - page_offset);
        
        // 查找缓存页
        struct cached_page *page = find_cached_page(inode->inode_number, page_index);
        
        if (!page) {
            // 缓存未命中，从磁盘读取
            page = allocate_cache_page(inode->inode_number, page_index);
            read_blocks_from_disk(get_blocks_for_page(inode, page_index), 
                                 page->data, PAGE_SIZE / BLOCK_SIZE);
        }
        
        // 从缓存复制数据
        memcpy(buf + bytes_read, page->data + page_offset, bytes_in_page);
        
        bytes_read += bytes_in_page;
        mark_page_accessed(page);  // 更新LRU
    }
    
    return bytes_read;
}

struct journal_handle *journal_start(struct journal *journal, int num_blocks) {

        handle->transaction = create_new_transaction(journal);
    }
    
    list_add(&handle->buffers, &handle->transaction->handles);
    handle->buffer_count = 0;
    handle->error_code = 0;
    
    return handle;
}

10.2 日志写入流程 ​

c
int journal_write_metadata(struct journal_handle *handle, 
                          struct buffer_head *bh) {
    // 1. 将元数据写入日志
    struct journal_block *jblock = allocate_journal_block(handle->transaction);
    memcpy(jblock->data, bh->b_data, BLOCK_SIZE);
    
    // 2. 记录缓冲区与日志块的映射
    bh->b_transaction = handle->transaction;
    bh->b_jblock = jblock;
    
    // 3. 增加事务引用计数
    handle->buffer_count++;
    
    return 0;
}

// 提交事务
int journal_stop(struct journal_handle *handle) {
    struct journal_transaction *transaction = handle->transaction;
    
    // 1. 等待所有元数据写入日志
    wait_for_metadata_writeback(transaction);
    
    // 2. 写入提交记录
    write_commit_record(transaction);
    
    // 3. 将日志内容写回文件系统
    writeback_dirty_buffers(transaction);
    
    // 4. 释放事务
    if (list_empty(&transaction->handles)) {
        free_transaction(transaction);
    }
    
    kfree(handle);
    return 0;
}

11.1 写时复制（Copy-on-Write） ​

c
// Btrfs/ZFS使用的写时复制机制
struct cow_write_operation {
    struct inode *inode;
    uint64_t file_offset;
    uint32_t length;
    const void *data;
    
    // 新分配的块
    uint32_t new_blocks[16];
    uint32_t new_block_count;
};

int cow_write(struct cow_write_operation *op) {
    // 1. 为新数据分配块（不覆盖旧块）
    for (int i = 0; i < op->length; i += BLOCK_SIZE) {
        uint32_t new_block = allocate_block();
        op->new_blocks[op->new_block_count++] = new_block;
        
        // 复制数据到新块
        write_block(new_block, op->data + i, MIN(BLOCK_SIZE, op->length - i));
    }
    
    // 2. 原子性地更新指针
    update_file_extents(op->inode, op->file_offset, op->new_blocks, op->new_block_count);
    
    // 3. 旧块可以被垃圾回收
    add_blocks_to_garbage_collection(op->inode, op->file_offset, op->length);
    
    return 0;
}

12. 性能优化技术 ​

12.1 预读（Read-ahead） ​

c
struct readahead_control {
    struct file *file;
    pgoff_t index;          // 当前页面索引
    pgoff_t prev_index;     // 上一次读取的页面索引
    unsigned long ra_pages; // 预读页面数
    unsigned long size;     // 预读窗口大小
};

void do_page_cache_readahead(struct readahead_control *rac) {
    pgoff_t index = rac->index;
    unsigned long max_pages = rac->ra_pages;
    int hit_readahead_marker = 0;
    
    // 计算预读大小
    unsigned long expected = rac->prev_index + 1;
    if (index == expected) {
        // 顺序读取，增加预读窗口
        rac->size = min(rac->size * 2, max_pages);
    } else {
        // 随机读取，重置预读窗口
        rac->size = 1;
    }
    
    // 异步读取页面
    for (int i = 0; i < rac->size; i++) {
        struct page *page = find_or_create_page(rac->file->f_mapping, index + i);
        if (!page || !trylock_page(page)) continue;
        
        if (PageUptodate(page)) {
            unlock_page(page);
            continue;
        }
        
        // 提交读取请求
        submit_page_read(page);
        unlock_page(page);
    }
}

int filesystem_check(struct superblock *sb) {
    int errors_found = 0;
    
    // 1. 检查超级块
    if (!check_superblock(sb)) {
        errors_found++;
        repair_superblock(sb);
    }
    
    // 2. 检查inode一致性
    errors_found += check_inode_consistency(sb);
    
    // 3. 检查目录结构
    errors_found += check_directory_structure(sb);
    
    // 4. 检查数据块分配
    errors_found += check_block_allocation(sb);
    
    // 5. 修复发现的错误
    if (errors_found > 0) {
        repair_filesystem(sb);
    }
    
    return errors_found;
}

15. 总结 ​

文件系统是操作系统的核心组件，它提供了：

  1. 数据组织：通过目录树和文件结构组织数据
  2. 持久化存储：将数据可靠地保存在存储设备上
  3. 访问控制：管理用户权限和数据安全
  4. 性能优化：通过缓存、预读等技术提高IO效率
  5. 数据一致性：通过日志、写时复制等机制保证数据完整性

现代文件系统正朝着更高效、更可靠、功能更丰富的方向发展，如分布式文件系统、加密文件系统等，持续满足日益增长的数据存储需求。
