操作系统 I/O 管理 - I/O 多路复用学习笔记 ​

1. I/O 多路复用概述 ​

1.1 基本概念 ​

I/O 多路复用是一种使单个进程能够同时监视多个文件描述符（通常是网络套接字）的机制，当其中任何一个文件描述符就绪（可读、可写或发生异常）时，程序能够立即得到通知。

1.2 解决的问题 ​

  • 避免为每个I/O连接创建线程/进程的开销
  • 提高服务器并发处理能力
  • 减少系统资源消耗
  • 简化并发编程模型

2. I/O 模型对比 ​

2.1 阻塞 I/O ​

c
// 传统阻塞方式
int client_fd = accept(server_fd, ...);  // 阻塞等待连接
read(client_fd, buffer, size);           // 阻塞等待数据

缺点：每个连接需要单独的线程/进程处理

2.2 非阻塞 I/O ​

c
// 设置非阻塞
fcntl(fd, F_SETFL, O_NONBLOCK);

// 轮询检查
while (true) {
    for (每个fd) {
        if (read(fd, buffer, size) > 0) {
            // 处理数据
        }
    }
}

缺点：CPU 空转，效率低下

2.3 I/O 多路复用 ​

c
// 使用 select/poll/epoll 监视多个fd
while (true) {
    ready_count = epoll_wait(epoll_fd, events, MAX_EVENTS, -1);
    for (i = 0; i < ready_count; i++) {
        // 处理就绪的fd
    }
}

优点：高效，CPU利用率高

3. select 系统调用 ​

3.1 函数原型 ​

c
#include <sys/select.h>

int select(int nfds, 
          fd_set *readfds, 
          fd_set *writefds,
          fd_set *exceptfds, 
          struct timeval *timeout);

3.2 核心数据结构 ​

c
// fd_set 操作宏
FD_ZERO(fd_set *set);      // 清空集合
FD_SET(int fd, fd_set *set);   // 添加fd到集合
FD_CLR(int fd, fd_set *set);   // 从集合移除fd
FD_ISSET(int fd, fd_set *set); // 检查fd是否在集合中

3.3 使用示例 ​

c
#include <stdio.h>
#include <stdlib.h>
#include <sys/select.h>
#include <sys/socket.h>

#define MAX_CLIENTS 10

int main() {
    fd_set read_fds;
    int client_sockets[MAX_CLIENTS] = {0};
    int max_fd, activity, i;
    
    while (1) {
        FD_ZERO(&read_fds);
        
        // 添加服务器socket
        FD_SET(server_socket, &read_fds);
        max_fd = server_socket;
        
        // 添加客户端sockets
        for (i = 0; i < MAX_CLIENTS; i++) {
            if (client_sockets[i] > 0) {
                FD_SET(client_sockets[i], &read_fds);
                if (client_sockets[i] > max_fd)
                    max_fd = client_sockets[i];
            }
        }
        
        // 等待活动
        activity = select(max_fd + 1, &read_fds, NULL, NULL, NULL);
        
        if (activity < 0) {
            perror("select error");
            continue;
        }
        
        // 检查新连接
        if (FD_ISSET(server_socket, &read_fds)) {
            accept_new_connection(server_socket, client_sockets);
        }
        
        // 检查客户端数据
        for (i = 0; i < MAX_CLIENTS; i++) {
            if (client_sockets[i] > 0 && 
                FD_ISSET(client_sockets[i], &read_fds)) {
                handle_client_data(client_sockets[i], i);
            }
        }
    }
    return 0;
}

3.4 select 的局限性 ​

  • 文件描述符数量限制（通常 1024）
  • 每次调用需要重新设置fd_set
  • 需要遍历所有fd来检查状态
  • 性能随fd数量增加而下降

4. poll 系统调用 ​

4.1 函数原型 ​

c
#include <poll.h>

int poll(struct pollfd *fds, 
         nfds_t nfds, 
         int timeout);

4.2 核心数据结构 ​

c
struct pollfd {
    int fd;           // 文件描述符
    short events;     // 等待的事件
    short revents;    // 实际发生的事件
};

// 事件标志
POLLIN   - 数据可读
POLLOUT  - 数据可写
POLLERR  - 发生错误
POLLHUP  - 连接挂起

4.3 使用示例 ​

c
#include <poll.h>

#define MAX_CLIENTS 1000

int main() {
    struct pollfd fds[MAX_CLIENTS];
    int nfds = 1;  // 初始只有服务器socket
    int timeout = 5000; // 5秒超时
    
    // 初始化服务器socket
    fds[0].fd = server_socket;
    fds[0].events = POLLIN;
    
    while (1) {
        int ret = poll(fds, nfds, timeout);
        
        if (ret < 0) {
            perror("poll error");
            break;
        }
        
        if (ret == 0) {
            printf("Timeout occurred\n");
            continue;
        }
        
        // 检查所有fd
        for (int i = 0; i < nfds; i++) {
            if (fds[i].revents == 0)
                continue;
                
            if (fds[i].revents & POLLIN) {
                if (fds[i].fd == server_socket) {
                    // 接受新连接
                    accept_connection(fds, &nfds, server_socket);
                } else {
                    // 处理客户端数据
                    handle_client(fds[i].fd);
                }
            }
            
            if (fds[i].revents & (POLLERR | POLLHUP)) {
                // 处理错误或连接关闭
                close(fds[i].fd);
                remove_fd(fds, &nfds, i);
            }
        }
    }
    return 0;
}

4.4 poll 的优点 ​

  • 没有文件描述符数量限制
  • 不需要每次重新初始化数据结构
  • 更清晰的事件分离（events/revents）

4.5 poll 的缺点 ​

  • 仍然需要遍历所有fd
  • 性能在大规模连接时下降
  • 不是Linux内核最优实现

5. epoll 系统调用 ​

5.1 epoll 的核心优势 ​

  • 事件驱动：只返回就绪的文件描述符
  • **水平触发(LT)和边缘触发(ET)**模式
  • 内存映射：减少内核与用户空间数据拷贝
  • 可扩展性：支持数十万并发连接

5.2 相关系统调用 ​

c
#include <sys/epoll.h>

// 创建epoll实例
int epoll_create(int size);
int epoll_create1(int flags);

// 控制epoll事件
int epoll_ctl(int epfd, int op, int fd, struct epoll_event *event);

// 等待事件
int epoll_wait(int epfd, struct epoll_event *events,
               int maxevents, int timeout);

5.3 核心数据结构 ​

c
typedef union epoll_data {
    void *ptr;
    int fd;
    uint32_t u32;
    uint64_t u64;
} epoll_data_t;

struct epoll_event {
    uint32_t events;      // Epoll事件
    epoll_data_t data;    // 用户数据
};

// 事件类型
EPOLLIN   - 可读
EPOLLOUT  - 可写  
EPOLLET   - 边缘触发模式
EPOLLONESHOT - 一次性事件
EPOLLRDHUP - 对端关闭连接

5.4 使用示例 ​

c
#include <sys/epoll.h>

#define MAX_EVENTS 64

int main() {
    int epoll_fd, nfds, i;
    struct epoll_event ev, events[MAX_EVENTS];
    
    // 创建epoll实例
    epoll_fd = epoll_create1(0);
    if (epoll_fd == -1) {
        perror("epoll_create1");
        exit(EXIT_FAILURE);
    }
    
    // 添加服务器socket到epoll
    ev.events = EPOLLIN;
    ev.data.fd = server_socket;
    if (epoll_ctl(epoll_fd, EPOLL_CTL_ADD, server_socket, &ev) == -1) {
        perror("epoll_ctl: server_socket");
        exit(EXIT_FAILURE);
    }
    
    while (1) {
        nfds = epoll_wait(epoll_fd, events, MAX_EVENTS, -1);
        if (nfds == -1) {
            perror("epoll_wait");
            exit(EXIT_FAILURE);
        }
        
        for (i = 0; i < nfds; i++) {
            if (events[i].data.fd == server_socket) {
                // 处理新连接
                accept_connection(epoll_fd, server_socket);
            } else {
                // 处理客户端请求
                handle_client_request(events[i].data.fd, epoll_fd);
            }
        }
    }
    
    close(epoll_fd);
    return 0;
}

5.5 水平触发 vs 边缘触发 ​

水平触发 (Level-Triggered, LT) ​

  • 默认模式
  • 只要文件描述符就绪，就会持续通知
  • 编程更简单，不容易丢失事件
c
// LT模式示例
ev.events = EPOLLIN;  // 默认LT模式

边缘触发 (Edge-Triggered, ET) ​

  • 需要显式设置
  • 只在状态变化时通知一次
  • 性能更好，但编程更复杂
  • 必须使用非阻塞I/O
c
// ET模式示例
ev.events = EPOLLIN | EPOLLET;  // 边缘触发模式

// ET模式下必须非阻塞IO
fcntl(fd, F_SETFL, fcntl(fd, F_GETFL, 0) | O_NONBLOCK);

5.6 ET模式完整示例 ​

c
void handle_et_client(int client_fd, int epoll_fd) {
    char buffer[1024];
    ssize_t count;
    
    while (1) {
        count = read(client_fd, buffer, sizeof(buffer));
        
        if (count == -1) {
            // 对于非阻塞IO，EAGAIN表示数据已读完
            if (errno == EAGAIN || errno == EWOULDBLOCK) {
                // 数据读取完毕，重新添加EPOLLIN事件
                struct epoll_event ev;
                ev.events = EPOLLIN | EPOLLET;
                ev.data.fd = client_fd;
                epoll_ctl(epoll_fd, EPOLL_CTL_MOD, client_fd, &ev);
                break;
            }
            // 真实错误
            close(client_fd);
            break;
        } else if (count == 0) {
            // 连接关闭
            close(client_fd);
            break;
        } else {
            // 处理数据
            process_data(buffer, count);
        }
    }
}

6. 性能对比分析 ​

6.1 时间复杂度 ​

方法添加fd删除fd等待事件
selectO(n)O(n)O(n)
pollO(1)O(1)O(n)
epollO(1)O(1)O(1)

6.2 适用场景 ​

  • select: 跨平台，fd数量少(<1024)
  • poll: fd数量中等，需要跨Unix平台
  • epoll: Linux平台，高性能要求，大量并发连接

7. 实际应用模式 ​

7.1 Reactor 模式 ​

c
// Reactor核心组件
typedef struct {
    int epoll_fd;
    event_handler_t **handlers;
    int max_events;
} reactor_t;

typedef struct {
    int fd;
    void (*handle_event)(int fd, uint32_t events);
} event_handler_t;

void reactor_run(reactor_t *reactor) {
    struct epoll_event *events = malloc(
        reactor->max_events * sizeof(struct epoll_event));
    
    while (1) {
        int n = epoll_wait(reactor->epoll_fd, events, 
                          reactor->max_events, -1);
        for (int i = 0; i < n; i++) {
            event_handler_t *handler = 
                (event_handler_t*)events[i].data.ptr;
            handler->handle_event(handler->fd, events[i].events);
        }
    }
    free(events);
}

7.2 连接管理最佳实践 ​

c
#define MAX_EPOLL_EVENTS 4096

typedef struct {
    int fd;
    char read_buf[8192];
    char write_buf[8192];
    size_t read_len;
    size_t write_len;
} connection_t;

void setup_epoll_server() {
    int epoll_fd = epoll_create1(0);
    struct epoll_event event;
    connection_t *conns[MAX_EPOLL_EVENTS] = {0};
    
    // 设置服务器socket
    int server_fd = create_server_socket();
    set_nonblocking(server_fd);
    
    event.events = EPOLLIN | EPOLLET;
    event.data.fd = server_fd;
    epoll_ctl(epoll_fd, EPOLL_CTL_ADD, server_fd, &event);
    
    struct epoll_event events[MAX_EPOLL_EVENTS];
    
    while (1) {
        int n = epoll_wait(epoll_fd, events, MAX_EPOLL_EVENTS, -1);
        
        for (int i = 0; i < n; i++) {
            if (events[i].data.fd == server_fd) {
                // 接受所有待处理连接
                while (1) {
                    int client_fd = accept(server_fd, NULL, NULL);
                    if (client_fd == -1) {
                        if (errno == EAGAIN || errno == EWOULDBLOCK) {
                            break; // 所有连接处理完毕
                        }
                        perror("accept");
                        break;
                    }
                    setup_client_connection(client_fd, epoll_fd, conns);
                }
            } else {
                handle_client_event(events[i], epoll_fd, conns);
            }
        }
    }
}

8. 常见问题与解决方案 ​

8.1 惊群问题 ​

多个进程/线程等待同一个socket，当连接到来时所有等待者都被唤醒。

解决方案：

  • 使用 EPOLLEXCLUSIVE (Linux 4.5+)
  • 使用SO_REUSEPORT
  • 应用层负载均衡

8.2 文件描述符耗尽 ​

c
// 监控和限制连接数
void check_fd_limits() {
    struct rlimit limits;
    getrlimit(RLIMIT_NOFILE, &limits);
    printf("Max FDs: soft=%ld, hard=%ld\n", 
           limits.rlim_cur, limits.rlim_max);
}

// 优雅处理EMFILE错误
void handle_emfile(int epoll_fd) {
    // 临时接受并立即关闭连接
    int temp_fd = accept(server_fd, NULL, NULL);
    if (temp_fd != -1) {
        close(temp_fd);
    }
}

8.3 内存管理 ​

c
// 连接池管理
typedef struct {
    connection_t **pool;
    int size;
    int count;
} connection_pool_t;

connection_t *get_connection(connection_pool_t *pool, int fd) {
    if (pool->count >= pool->size) {
        // 动态扩容
        resize_pool(pool);
    }
    
    connection_t *conn = malloc(sizeof(connection_t));
    memset(conn, 0, sizeof(connection_t));
    conn->fd = fd;
    pool->pool[pool->count++] = conn;
    
    return conn;
}

io_uring_queue_init(ENTRIES, &ring, 0);

// 处理完成事件

10. 总结 ​

I/O多路复用是现代高性能网络编程的核心技术：

  1. select/poll 适用于小规模并发和跨平台场景
  2. epoll 是Linux下高性能网络服务的首选
  3. ET模式 提供最佳性能，但编程复杂度更高
  4. LT模式 更易使用，适合大多数场景
  5. 合理的内存管理和连接池设计至关重要
  6. 新的io_uring技术提供了更好的性能前景

掌握I/O多路复用技术对于构建高性能、高并发的网络服务至关重要。
