操作系统内存管理 - 分页机制完整学习笔记 ​

1. 分页机制概述 ​

1.1 基本概念 ​

  • 分页：将进程的地址空间和物理内存划分为固定大小的块
  • 页面：进程地址空间中的固定大小块（逻辑单元）
  • 页框：物理内存中的固定大小块（物理单元）
  • 页表：存储页面到页框映射关系的数据结构

1.2 分页解决的问题 ​

  • 外部碎片问题
  • 内存分配效率
  • 进程隔离与保护
  • 虚拟内存实现基础

2. 分页基本原理 ​

2.1 地址划分 ​

虚拟地址结构：
+----------------+----------------+
|    页号(p)     |  页内偏移(d)   |
+----------------+----------------+
  • 页号：标识虚拟页面
  • 页内偏移：在页面内的字节位置

2.2 地址转换过程 ​

虚拟地址 → [页表查询] → 物理地址
页号(p) + 偏移(d) → 帧号(f) + 偏移(d)

2.3 页表结构 ​

c
// 典型的页表项结构
struct page_table_entry {
    uint32_t present : 1;      // 页面是否在内存中
    uint32_t rw : 1;           // 读写权限
    uint32_t user : 1;         // 用户/内核模式
    uint32_t accessed : 1;     // 访问标记
    uint32_t dirty : 1;        // 修改标记
    uint32_t frame_number : 20; // 物理帧号
    uint32_t reserved : 7;     // 保留位
};

3. 页表设计与实现 ​

3.1 简单线性页表 ​

c
// 简单的线性页表实现
#define PAGE_SIZE 4096
#define NUM_PAGES 1024

struct page_table {
    struct page_table_entry entries[NUM_PAGES];
};

// 地址转换函数
physical_addr_t translate_address(virtual_addr_t va, struct page_table *pt) {
    uint32_t page_num = va / PAGE_SIZE;
    uint32_t offset = va % PAGE_SIZE;
    
    if (!pt->entries[page_num].present) {
        handle_page_fault(page_num);  // 处理缺页异常
    }
    
    return (pt->entries[page_num].frame_number * PAGE_SIZE) + offset;
}

3.2 多级页表 ​

两级页表结构 ​

虚拟地址：
+-----------+-----------+-----------+
| 页目录索引 | 页表索引  |  偏移量   |
+-----------+-----------+-----------+

实现示例 ​

c
#define PAGE_DIR_ENTRIES 1024
#define PAGE_TABLE_ENTRIES 1024

struct page_directory_entry {
    uint32_t present : 1;
    uint32_t rw : 1;
    uint32_t user : 1;
    uint32_t page_table_base : 20; // 页表物理地址高20位
    uint32_t reserved : 9;
};

struct page_directory {
    struct page_directory_entry entries[PAGE_DIR_ENTRIES];
};

// 两级地址转换
physical_addr_t two_level_translate(virtual_addr_t va, struct page_directory *pd) {
    uint32_t dir_index = (va >> 22) & 0x3FF;    // 高10位
    uint32_t table_index = (va >> 12) & 0x3FF;  // 中间10位
    uint32_t offset = va & 0xFFF;               // 低12位
    
    struct page_directory_entry *pde = &pd->entries[dir_index];
    if (!pde->present) {
        handle_page_fault_dir(dir_index);
    }
    
    struct page_table *pt = (struct page_table*)(pde->page_table_base << 12);
    struct page_table_entry *pte = &pt->entries[table_index];
    
    if (!pte->present) {
        handle_page_fault_table(table_index);
    }
    
    return (pte->frame_number * PAGE_SIZE) + offset;
}

4. 转换检测缓冲区(TLB) ​

4.1 TLB工作原理 ​

  • 缓存最近使用的页表项
  • 减少内存访问次数
  • 基于局部性原理

4.2 TLB管理 ​

c
#define TLB_ENTRIES 64

struct tlb_entry {
    uint32_t virtual_page;
    uint32_t physical_frame;
    uint32_t valid : 1;
    uint32_t asid : 8;  // 地址空间标识符
};

struct tlb {
    struct tlb_entry entries[TLB_ENTRIES];
    uint32_t next_replace;  // 替换指针
};

// TLB查找
int tlb_lookup(struct tlb *tlb, uint32_t virtual_page, uint32_t asid) {
    for (int i = 0; i < TLB_ENTRIES; i++) {
        if (tlb->entries[i].valid && 
            tlb->entries[i].virtual_page == virtual_page &&
            tlb->entries[i].asid == asid) {
            return tlb->entries[i].physical_frame;
        }
    }
    return -1;  // TLB未命中
}

// TLB更新
void tlb_update(struct tlb *tlb, uint32_t virtual_page, 
                uint32_t physical_frame, uint32_t asid) {
    // 简单FIFO替换
    tlb->entries[tlb->next_replace].virtual_page = virtual_page;
    tlb->entries[tlb->next_replace].physical_frame = physical_frame;
    tlb->entries[tlb->next_replace].asid = asid;
    tlb->entries[tlb->next_replace].valid = 1;
    
    tlb->next_replace = (tlb->next_replace + 1) % TLB_ENTRIES;
}

6. 页面分配与回收 ​

6.1 帧分配算法 ​

c
// 帧分配数据结构
struct frame_allocator {
    uint32_t *free_frames;      // 空闲帧位图
    uint32_t total_frames;      // 总帧数
    uint32_t free_count;        // 空闲帧计数
    spinlock_t lock;           // 保护锁
};

// 分配物理帧
int allocate_frame(struct frame_allocator *alloc) {
    spin_lock(&alloc->lock);
    
    if (alloc->free_count == 0) {
        spin_unlock(&alloc->lock);
        return -1;  // 内存不足
    }
    
    // 查找第一个空闲帧
    for (int i = 0; i < alloc->total_frames; i++) {
        if (alloc->free_frames[i / 32] & (1 << (i % 32))) {
            alloc->free_frames[i / 32] &= ~(1 << (i % 32));
            alloc->free_count--;
            spin_unlock(&alloc->lock);
            return i;
        }
    }
    
    spin_unlock(&alloc->lock);
    return -1;
}

// 释放物理帧
void free_frame(struct frame_allocator *alloc, int frame) {
    spin_lock(&alloc->lock);
    alloc->free_frames[frame / 32] |= (1 << (frame % 32));
    alloc->free_count++;
    spin_unlock(&alloc->lock);
}

6.2 分配策略 ​

  • 均衡分配：所有进程平均分配帧
  • 比例分配：按进程大小比例分配
  • 优先级分配：高优先级进程获得更多帧

7. 页面置换算法 ​

7.1 页面置换框架 ​

c
struct page_replacement {
    uint32_t *frames;           // 帧数组
    uint32_t num_frames;        // 帧数量
    uint32_t *access_bits;      // 访问位
    uint32_t *dirty_bits;       // 修改位
    uint64_t *access_time;      // 访问时间
};

// 通用置换接口
typedef uint32_t (*replacement_func_t)(struct page_replacement *pr);

7.2 常见置换算法实现 ​

FIFO算法 ​

c
uint32_t fifo_replacement(struct page_replacement *pr) {
    static uint32_t pointer = 0;
    uint32_t victim = pointer;
    pointer = (pointer + 1) % pr->num_frames;
    return victim;
}

LRU算法（近似） ​

c
uint32_t lru_approx_replacement(struct page_replacement *pr) {
    uint32_t victim = 0;
    uint64_t oldest = pr->access_time[0];
    
    for (uint32_t i = 1; i < pr->num_frames; i++) {
        if (pr->access_time[i] < oldest) {
            oldest = pr->access_time[i];
            victim = i;
        }
    }
    return victim;
}

时钟算法 ​

c
uint32_t clock_replacement(struct page_replacement *pr) {
    static uint32_t hand = 0;
    
    while (1) {
        if (pr->access_bits[hand] == 0) {
            uint32_t victim = hand;
            hand = (hand + 1) % pr->num_frames;
            return victim;
        } else {
            pr->access_bits[hand] = 0;  // 给第二次机会
            hand = (hand + 1) % pr->num_frames;
        }
    }
}

8. 缺页异常处理 ​

8.1 缺页处理流程 ​

c
void page_fault_handler(virtual_addr_t fault_addr, uint32_t error_code) {
    // 1. 检查地址合法性
    if (!is_valid_address(current_process, fault_addr)) {
        kill_process(current_process, SIGSEGV);
        return;
    }
    
    // 2. 检查权限
    if ((error_code & PAGE_FAULT_WRITE) && 
        !is_writable(fault_addr)) {
        kill_process(current_process, SIGSEGV);
        return;
    }
    
    // 3. 分配物理帧
    uint32_t frame = allocate_frame(&frame_allocator);
    if (frame == -1) {
        // 内存不足，需要页面置换
        frame = perform_page_replacement();
    }
    
    // 4. 从交换空间加载数据
    load_page_from_swap(fault_addr, frame);
    
    // 5. 更新页表
    update_page_table(current_process, fault_addr, frame);
    
    // 6. 刷新TLB
    flush_tlb_entry(fault_addr);
}

9. 共享内存与写时复制 ​

9.1 共享页面 ​

c
struct shared_page {
    uint32_t physical_frame;
    uint32_t reference_count;
    struct list_head user_processes;  // 使用该页的进程列表
};

// 共享页面映射
int map_shared_page(struct process *p, virtual_addr_t va, 
                   struct shared_page *sp) {
    // 更新页表，指向共享物理帧
    update_page_table(p, va, sp->physical_frame);
    
    // 设置只读权限，用于写时复制
    set_page_readonly(p, va);
    
    // 增加引用计数
    sp->reference_count++;
    
    return 0;
}

9.2 写时复制处理 ​

c
void handle_cow_fault(virtual_addr_t fault_addr) {
    // 1. 获取原始页面
    uint32_t old_frame = get_physical_frame(fault_addr);
    struct shared_page *sp = find_shared_page(old_frame);
    
    // 2. 分配新帧
    uint32_t new_frame = allocate_frame(&frame_allocator);
    
    // 3. 复制页面内容
    memcpy((void*)(new_frame * PAGE_SIZE), 
           (void*)(old_frame * PAGE_SIZE), 
           PAGE_SIZE);
    
    // 4. 更新页表指向新帧
    update_page_table(current_process, fault_addr, new_frame);
    set_page_writable(current_process, fault_addr);
    
    // 5. 减少共享页面引用计数
    sp->reference_count--;
    if (sp->reference_count == 0) {
        free_shared_page(sp);
    }
}

12. 总结 ​

分页机制是现代操作系统内存管理的核心技术，它通过：

  • 消除外部碎片
  • 提供进程隔离
  • 实现虚拟内存
  • 支持共享内存

关键优化技术包括TLB、多级页表、页面置换算法等，这些技术共同保证了内存管理系统的高效性和可靠性。

这份笔记涵盖了分页机制的核心概念、实现细节和优化技术，可作为深入理解操作系统内存管理的基础。
