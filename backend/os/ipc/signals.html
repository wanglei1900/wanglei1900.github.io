操作系统知识 - 进程间通信 - 信号 ​

1. 信号的基本概念 ​

1.1 什么是信号 ​

  • 信号是Linux/Unix系统中进程间通信的一种异步通知机制
  • 用于通知进程发生了某种事件，类似于软件中断
  • 信号是操作系统内核管理的一种资源

1.2 信号的特点 ​

  • 异步性：信号可以在任何时候发送给进程
  • 简单性：只能传递信号类型，不能携带复杂数据
  • 不可靠性：早期信号可能丢失，现代信号已解决此问题
  • 优先级：不同信号有不同的优先级

2. 信号的产生方式 ​

2.1 硬件异常产生的信号 ​

c
// 示例：除零错误产生 SIGFPE 信号
int main() {
    int a = 10, b = 0;
    int c = a / b;  // 产生 SIGFPE 信号
    return 0;
}

2.2 终端特殊按键产生的信号 ​

  • Ctrl+C：产生 SIGINT 信号（中断）
  • Ctrl+\：产生 SIGQUIT 信号（退出）
  • Ctrl+Z：产生 SIGTSTP 信号（暂停）

2.3 软件条件产生的信号 ​

c
// 示例：alarm函数产生SIGALRM信号
#include <unistd.h>
#include <signal.h>

void alarm_handler(int sig) {
    write(1, "Alarm!\n", 7);
}

int main() {
    signal(SIGALRM, alarm_handler);
    alarm(5);  // 5秒后产生SIGALRM信号
    pause();   // 等待信号
    return 0;
}

3. 信号的分类 ​

3.1 不可靠信号（1-31） ​

  • 早期Unix系统中的信号，可能丢失
  • 不支持排队，相同信号可能被合并

3.2 可靠信号（34-64） ​

  • Linux扩展的信号，支持排队
  • 不会丢失

3.3 常见信号列表 ​

信号编号信号名称默认动作说明
1SIGHUP终止终端挂起或控制进程终止
2SIGINT终止键盘中断（Ctrl+C）
3SIGQUIT终止+core键盘退出（Ctrl+\）
4SIGILL终止+core非法指令
8SIGFPE终止+core浮点异常
9SIGKILL终止强制杀死进程（不可捕获）
11SIGSEGV终止+core段错误
14SIGALRM终止定时器信号
15SIGTERM终止程序结束信号
17SIGCHLD忽略子进程状态改变
18SIGCONT继续继续执行暂停的进程
19SIGSTOP暂停暂停进程（不可捕获）
20SIGTSTP暂停终端暂停（Ctrl+Z）

4. 信号的处理方式 ​

4.1 默认处理 ​

  • 执行系统默认动作（终止、终止+core、忽略、暂停、继续）

4.2 忽略信号 ​

c
#include <signal.h>
#include <stdio.h>

int main() {
    // 忽略SIGINT信号（Ctrl+C无效）
    signal(SIGINT, SIG_IGN);
    
    while(1) {
        printf("Running... Press Ctrl+C has no effect.\n");
        sleep(1);
    }
    return 0;
}

4.3 捕获信号（自定义处理函数） ​

c
#include <signal.h>
#include <stdio.h>
#include <unistd.h>

// 信号处理函数
void sigint_handler(int sig) {
    printf("\nReceived SIGINT signal %d\n", sig);
    printf("Do you really want to exit? (y/n): ");
    
    char answer;
    scanf("%c", &answer);
    if (answer == 'y' || answer == 'Y') {
        exit(0);
    }
}

int main() {
    // 注册信号处理函数
    signal(SIGINT, sigint_handler);
    
    while(1) {
        printf("Program running... Press Ctrl+C to test.\n");
        sleep(2);
    }
    return 0;
}

5. 信号集操作 ​

5.1 信号集数据类型 ​

c
#include <signal.h>

// 信号集操作函数
int sigemptyset(sigset_t *set);      // 清空信号集
int sigfillset(sigset_t *set);       // 填充所有信号
int sigaddset(sigset_t *set, int signo);    // 添加信号
int sigdelset(sigset_t *set, int signo);    // 删除信号
int sigismember(const sigset_t *set, int signo); // 测试信号是否在集合中

5.2 信号阻塞 ​

c
#include <signal.h>
#include <stdio.h>
#include <unistd.h>

int main() {
    sigset_t newset, oldset;
    
    // 初始化信号集
    sigemptyset(&newset);
    sigaddset(&newset, SIGINT);  // 阻塞SIGINT
    
    // 设置信号屏蔽字
    sigprocmask(SIG_BLOCK, &newset, &oldset);
    
    printf("SIGINT is blocked for 10 seconds. Try Ctrl+C...\n");
    sleep(10);
    
    // 恢复原来的信号屏蔽字
    sigprocmask(SIG_SETMASK, &oldset, NULL);
    printf("SIGINT unblocked. Now try Ctrl+C...\n");
    
    sleep(10);
    return 0;
}

6. 高级信号处理 ​

6.1 sigaction函数 ​

c
#include <signal.h>
#include <stdio.h>
#include <unistd.h>

void signal_handler(int sig, siginfo_t *info, void *context) {
    printf("Received signal %d from process %d\n", 
           sig, info->si_pid);
    printf("User ID: %d\n", info->si_uid);
}

int main() {
    struct sigaction sa;
    
    sa.sa_sigaction = signal_handler;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = SA_SIGINFO;  // 使用sa_sigaction而不是sa_handler
    
    // 注册信号处理
    sigaction(SIGINT, &sa, NULL);
    
    printf("Process PID: %d\n", getpid());
    printf("Send SIGINT to test...\n");
    
    pause();  // 等待信号
    return 0;
}

6.2 可靠的信号处理 ​

c
#include <signal.h>
#include <stdio.h>
#include <unistd.h>
#include <string.h>

volatile sig_atomic_t signal_received = 0;

void handler(int sig) {
    signal_received = sig;
}

int main() {
    struct sigaction sa;
    
    memset(&sa, 0, sizeof(sa));
    sa.sa_handler = handler;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = SA_RESTART;  // 系统调用被信号中断后自动重启
    
    sigaction(SIGINT, &sa, NULL);
    sigaction(SIGTERM, &sa, NULL);
    
    while(1) {
        if (signal_received) {
            printf("Received signal: %d\n", signal_received);
            
            if (signal_received == SIGTERM) {
                printf("Graceful shutdown...\n");
                break;
            }
            signal_received = 0;
        }
        
        printf("Working...\n");
        sleep(1);
    }
    
    return 0;
}

7. 信号的发送 ​

7.1 kill函数 ​

c
#include <signal.h>
#include <sys/types.h>

// 向指定进程发送信号
int kill(pid_t pid, int sig);

7.2 raise函数 ​

c
#include <signal.h>

// 向当前进程发送信号
int raise(int sig);

7.3 示例：进程间信号通信 ​

c
// sender.c - 发送信号
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

int main(int argc, char *argv[]) {
    if (argc != 3) {
        printf("Usage: %s <pid> <signal>\n", argv[0]);
        exit(1);
    }
    
    pid_t pid = atoi(argv[1]);
    int sig = atoi(argv[2]);
    
    if (kill(pid, sig) == -1) {
        perror("kill");
        exit(1);
    }
    
    printf("Signal %d sent to process %d\n", sig, pid);
    return 0;
}

// receiver.c - 接收信号
#include <signal.h>
#include <stdio.h>
#include <unistd.h>

void handler(int sig) {
    printf("Process %d received signal %d\n", getpid(), sig);
}

int main() {
    printf("Receiver PID: %d\n", getpid());
    
    // 注册多个信号的处理函数
    signal(SIGUSR1, handler);
    signal(SIGUSR2, handler);
    signal(SIGTERM, handler);
    
    while(1) {
        pause();  // 等待信号
    }
    
    return 0;
}

8. 特殊信号处理 ​

8.1 SIGCHLD信号处理 ​

c
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>

void sigchld_handler(int sig) {
    int status;
    pid_t pid;
    
    // 使用WNOHANG避免阻塞
    while ((pid = waitpid(-1, &status, WNOHANG)) > 0) {
        if (WIFEXITED(status)) {
            printf("Child %d exited with status %d\n", 
                   pid, WEXITSTATUS(status));
        } else if (WIFSIGNALED(status)) {
            printf("Child %d killed by signal %d\n", 
                   pid, WTERMSIG(status));
        }
    }
}

int main() {
    struct sigaction sa;
    
    sa.sa_handler = sigchld_handler;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = SA_RESTART | SA_NOCLDSTOP;
    
    if (sigaction(SIGCHLD, &sa, NULL) == -1) {
        perror("sigaction");
        exit(1);
    }
    
    // 创建子进程
    pid_t pid = fork();
    if (pid == 0) {
        // 子进程
        printf("Child process %d started\n", getpid());
        sleep(3);
        printf("Child process %d exiting\n", getpid());
        exit(0);
    } else {
        // 父进程
        printf("Parent process %d, child PID: %d\n", getpid(), pid);
        printf("Parent waiting...\n");
        sleep(10);  // 等待SIGCHLD信号
        printf("Parent exiting\n");
    }
    
    return 0;
}

9. 信号的高级应用 ​

9.1 信号驱动I/O ​

c
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>

volatile sig_atomic_t data_ready = 0;

void io_handler(int sig) {
    data_ready = 1;
}

int main() {
    struct sigaction sa;
    int fd;
    
    // 设置信号处理
    sa.sa_handler = io_handler;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = 0;
    
    sigaction(SIGIO, &sa, NULL);
    
    // 打开文件并设置异步I/O
    fd = open("/dev/tty", O_RDWR);
    if (fd == -1) {
        perror("open");
        exit(1);
    }
    
    // 设置文件描述符的属主进程
    fcntl(fd, F_SETOWN, getpid());
    
    // 启用异步I/O
    fcntl(fd, F_SETFL, fcntl(fd, F_GETFL) | O_ASYNC);
    
    printf("Process %d waiting for I/O events...\n", getpid());
    
    while(1) {
        if (data_ready) {
            printf("I/O event occurred!\n");
            data_ready = 0;
            
            // 处理I/O
            char buffer[100];
            ssize_t n = read(fd, buffer, sizeof(buffer)-1);
            if (n > 0) {
                buffer[n] = '\0';
                printf("Read: %s", buffer);
            }
        }
        usleep(100000);  // 100ms
    }
    
    close(fd);
    return 0;
}

10. 信号的最佳实践和注意事项 ​

10.1 信号处理的安全规则 ​

  1. 保持处理函数简单：避免在信号处理函数中执行复杂操作
  2. 使用异步安全函数：在信号处理函数中只能使用异步安全函数
  3. 保存和恢复errno：保护全局变量不被破坏
  4. 处理可重入问题：避免使用非可重入函数

10.2 异步安全函数示例 ​

c
#include <signal.h>
#include <unistd.h>
#include <string.h>

// 异步安全的信号处理函数
void safe_handler(int sig) {
    // 只使用异步安全函数
    const char *msg = "Signal received\n";
    write(STDOUT_FILENO, msg, strlen(msg));
    
    // 可以安全地设置volatile sig_atomic_t变量
    // 但不能调用printf, malloc等不安全函数
}

10.3 完整的信号处理框架 ​

c
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <errno.h>

volatile sig_atomic_t shutdown_requested = 0;

void signal_handler(int sig) {
    int saved_errno = errno;  // 保存errno
    
    switch(sig) {
        case SIGINT:
            write(STDOUT_FILENO, "SIGINT received\n", 16);
            shutdown_requested = 1;
            break;
        case SIGTERM:
            write(STDOUT_FILENO, "SIGTERM received\n", 17);
            shutdown_requested = 1;
            break;
        case SIGUSR1:
            write(STDOUT_FILENO, "SIGUSR1 received\n", 17);
            break;
    }
    
    errno = saved_errno;  // 恢复errno
}

int setup_signal_handlers(void) {
    struct sigaction sa;
    
    sa.sa_handler = signal_handler;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = 0;
    
    // 阻塞其他信号 during signal handling
    sigaddset(&sa.sa_mask, SIGINT);
    sigaddset(&sa.sa_mask, SIGTERM);
    
    if (sigaction(SIGINT, &sa, NULL) == -1) return -1;
    if (sigaction(SIGTERM, &sa, NULL) == -1) return -1;
    if (sigaction(SIGUSR1, &sa, NULL) == -1) return -1;
    
    // 忽略某些信号
    signal(SIGPIPE, SIG_IGN);
    
    return 0;
}

int main() {
    if (setup_signal_handlers() == -1) {
        perror("setup_signal_handlers");
        exit(1);
    }
    
    printf("Process %d started. Send signals to test.\n", getpid());
    
    while(!shutdown_requested) {
        // 主循环工作
        printf("Working...\n");
        sleep(2);
    }
    
    printf("Graceful shutdown completed.\n");
    return 0;
}

总结 ​

信号是Linux/Unix系统中重要的进程间通信机制，具有以下关键点：

  1. 异步通知：信号提供了一种异步事件通知机制
  2. 简单通信：适合简单的通知，不适合复杂数据传递
  3. 多种处理方式：默认、忽略、自定义处理
  4. 可靠性改进：现代信号支持排队，解决了丢失问题
  5. 安全性考虑：信号处理函数需要特别注意线程安全和可重入性

掌握信号机制对于编写健壮的Linux/Unix应用程序至关重要。
