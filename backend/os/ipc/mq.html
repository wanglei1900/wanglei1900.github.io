操作系统知识 - 进程间通信：消息队列 ​

1. 消息队列概述 ​

1.1 基本概念 ​

消息队列 是操作系统提供的一种进程间通信机制，允许不相关的进程通过消息的形式进行数据交换。

1.2 主要特征 ​

  • 异步通信：发送者和接收者不需要同时存在
  • 消息边界：保持消息的完整性
  • 类型区分：可以按消息类型进行选择性接收
  • 内核持久性：消息队列由内核维护，独立于进程存在

2. 消息队列的工作原理 ​

2.1 基本结构 ​

发送进程 → 消息队列缓冲区 → 接收进程
        ↓
    内核空间

2.2 消息格式 ​

c
struct message {
    long mtype;     // 消息类型
    char mtext[1];  // 消息数据
};

3. System V 消息队列 ​

3.1 关键函数 ​

msgget - 创建/获取消息队列 ​

c
#include <sys/msg.h>

int msgget(key_t key, int msgflg);
  • key: 消息队列键值
  • msgflg: 权限标志
  • 返回：消息队列标识符

msgsnd - 发送消息 ​

c
int msgsnd(int msqid, const void *msgp, size_t msgsz, int msgflg);
  • msqid: 消息队列ID
  • msgp: 指向消息结构的指针
  • msgsz: 消息数据大小
  • msgflg: 标志位（IPC_NOWAIT等）

msgrcv - 接收消息 ​

c
ssize_t msgrcv(int msqid, void *msgp, size_t msgsz, long msgtyp, int msgflg);
  • msgtyp: 消息类型
    • = 0: 接收队列中第一条消息
    • > 0: 接收指定类型的消息
    • < 0: 接收类型 ≤ |msgtyp| 的最小消息

msgctl - 控制操作 ​

c
int msgctl(int msqid, int cmd, struct msqid_ds *buf);
  • cmd:
    • IPC_STAT: 获取状态
    • IPC_SET: 设置参数
    • IPC_RMID: 删除队列

3.2 示例代码 ​

发送进程 ​

c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/msg.h>
#include <unistd.h>

#define MAX_TEXT 512

struct my_msg {
    long int msg_type;
    char some_text[MAX_TEXT];
};

int main() {
    int running = 1;
    int msgid;
    struct my_msg some_data;
    char buffer[BUFSIZ];
    
    // 创建消息队列
    msgid = msgget((key_t)1234, 0666 | IPC_CREAT);
    if (msgid == -1) {
        perror("msgget failed");
        exit(EXIT_FAILURE);
    }
    
    while (running) {
        printf("Enter some text: ");
        fgets(buffer, BUFSIZ, stdin);
        some_data.msg_type = 1;
        strcpy(some_data.some_text, buffer);
        
        if (msgsnd(msgid, (void *)&some_data, MAX_TEXT, 0) == -1) {
            perror("msgsnd failed");
            exit(EXIT_FAILURE);
        }
        
        if (strncmp(buffer, "end", 3) == 0) {
            running = 0;
        }
    }
    exit(EXIT_SUCCESS);
}

接收进程 ​

c
#include <stdio.h>
#include <stdlib.h>
#include <sys/msg.h>
#include <unistd.h>

#define MAX_TEXT 512

struct my_msg {
    long int msg_type;
    char some_text[MAX_TEXT];
};

int main() {
    int running = 1;
    int msgid;
    struct my_msg some_data;
    long int msg_to_receive = 0;
    
    // 获取消息队列
    msgid = msgget((key_t)1234, 0666 | IPC_CREAT);
    if (msgid == -1) {
        perror("msgget failed");
        exit(EXIT_FAILURE);
    }
    
    while (running) {
        if (msgrcv(msgid, (void *)&some_data, BUFSIZ, 
                   msg_to_receive, 0) == -1) {
            perror("msgrcv failed");
            exit(EXIT_FAILURE);
        }
        printf("You wrote: %s", some_data.some_text);
        
        if (strncmp(some_data.some_text, "end", 3) == 0) {
            running = 0;
        }
    }
    
    // 删除消息队列
    if (msgctl(msgid, IPC_RMID, 0) == -1) {
        perror("msgctl(IPC_RMID) failed");
        exit(EXIT_FAILURE);
    }
    exit(EXIT_SUCCESS);
}

4. POSIX 消息队列 ​

4.1 关键函数 ​

mq_open - 打开/创建消息队列 ​

c
#include <mqueue.h>

mqd_t mq_open(const char *name, int oflag, mode_t mode, 
              struct mq_attr *attr);

mq_send - 发送消息 ​

c
int mq_send(mqd_t mqdes, const char *msg_ptr, 
            size_t msg_len, unsigned int msg_prio);

mq_receive - 接收消息 ​

c
ssize_t mq_receive(mqd_t mqdes, char *msg_ptr, 
                   size_t msg_len, unsigned int *msg_prio);

mq_close / mq_unlink - 关闭/删除 ​

c
int mq_close(mqd_t mqdes);
int mq_unlink(const char *name);

4.2 示例代码 ​

c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <mqueue.h>

#define QUEUE_NAME "/test_queue"
#define MAX_SIZE 1024
#define MSG_STOP "exit"

int main() {
    mqd_t mq;
    struct mq_attr attr;
    char buffer[MAX_SIZE];
    int must_stop = 0;
    
    // 设置消息队列属性
    attr.mq_flags = 0;
    attr.mq_maxmsg = 10;
    attr.mq_msgsize = MAX_SIZE;
    attr.mq_curmsgs = 0;
    
    // 创建消息队列
    mq = mq_open(QUEUE_NAME, O_CREAT | O_RDWR, 0644, &attr);
    if (mq == (mqd_t)-1) {
        perror("mq_open");
        exit(1);
    }
    
    printf("Send to server (enter \"exit\" to stop):\n");
    
    do {
        printf("> ");
        fflush(stdout);
        memset(buffer, 0, MAX_SIZE);
        fgets(buffer, MAX_SIZE, stdin);
        
        // 发送消息
        if (mq_send(mq, buffer, strlen(buffer)+1, 0) == -1) {
            perror("mq_send");
            break;
        }
        
        must_stop = (strncmp(buffer, MSG_STOP, strlen(MSG_STOP)) == 0);
    } while (!must_stop);
    
    mq_close(mq);
    mq_unlink(QUEUE_NAME);
    return 0;
}

5. 消息队列的属性管理 ​

5.1 System V 队列属性 ​

c
struct msqid_ds {
    struct ipc_perm msg_perm;     // 权限
    time_t          msg_stime;    // 最后发送时间
    time_t          msg_rtime;    // 最后接收时间
    time_t          msg_ctime;    // 最后修改时间
    unsigned long   msg_cbytes;   // 当前字节数
    msgqnum_t       msg_qnum;     // 当前消息数
    msglen_t        msg_qbytes;   // 最大字节数
    pid_t           msg_lspid;    // 最后发送进程PID
    pid_t           msg_lrpid;    // 最后接收进程PID
};

5.2 POSIX 队列属性 ​

c
struct mq_attr {
    long mq_flags;    // 阻塞标志
    long mq_maxmsg;   // 最大消息数
    long mq_msgsize;  // 最大消息大小
    long mq_curmsgs;  // 当前消息数
};

6. 消息队列的限制和配置 ​

6.1 系统限制 ​

bash
# 查看系统V消息队列限制
ipcs -l

# 查看POSIX消息队列限制
cat /proc/sys/fs/mqueue/msg_max     # 最大消息数
cat /proc/sys/fs/mqueue/msgsize_max # 最大消息大小

6.2 常用配置参数 ​

  • msgmax: 单个消息最大大小
  • msgmnb: 单个队列最大字节数
  • msgmni: 系统最大队列数

7. 消息队列的优势与劣势 ​

7.1 优势 ​

  1. 异步通信：发送和接收不需要同步
  2. 消息边界：保持消息完整性
  3. 类型过滤：可按类型选择性接收
  4. 内核持久：独立于进程生命周期
  5. 流量控制：内置缓冲区管理

7.2 劣势 ​

  1. 性能开销：需要内核态和用户态切换
  2. 容量限制：受系统配置限制
  3. 复杂性：比信号和管道更复杂
  4. 资源泄漏：需要显式清理

8. 应用场景 ​

8.1 适用场景 ​

  • 客户端-服务器架构
  • 任务队列系统
  • 日志收集系统
  • 分布式系统组件通信

8.2 不适用场景 ​

  • 实时性要求极高的通信
  • 大量小消息的传输
  • 简单的父子进程通信

9. 最佳实践 ​

9.1 设计建议 ​

  1. 合理设置消息大小：避免过小或过大
  2. 使用消息类型：实现优先级或路由
  3. 错误处理：处理队列满/空的情况
  4. 资源清理：确保队列被正确删除

9.2 性能优化 ​
