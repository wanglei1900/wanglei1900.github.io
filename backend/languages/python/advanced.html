函数式编程 ​

Lambda表达式 ​

python
# 匿名函数
square = lambda x: x ** 2
print(square(5))  # 25

# 配合高阶函数使用
numbers = [1, 2, 3, 4]
squared = list(map(lambda x: x**2, numbers))

Map/Filter/Reduce ​

python
# Map: 应用函数到每个元素
result = list(map(str.upper, ["hello", "world"])  # ['HELLO', 'WORLD']

# Filter: 过滤元素
evens = list(filter(lambda x: x % 2 == 0, range(10)))  # [0, 2, 4, 6, 8]

# Reduce: 累积计算
from functools import reduce
product = reduce(lambda x, y: x * y, [1, 2, 3, 4])  # 24

偏函数 (Partial) ​

python
from functools import partial

# 固定部分参数
def power(base, exponent):
    return base ** exponent

square = partial(power, exponent=2)
cube = partial(power, exponent=3)

print(square(5))  # 25
print(cube(3))    # 27

函数组合 ​

python
# 手动组合
def compose(f, g):
    return lambda x: f(g(x))

add_two = lambda x: x + 2
multiply_three = lambda x: x * 3

combined = compose(add_two, multiply_three)
print(combined(4))  # (4*3)+2 = 14

装饰器 ​

基础装饰器 ​

python
def simple_decorator(func):
    def wrapper():
        print("Before function call")
        func()
        print("After function call")
    return wrapper

@simple_decorator
def say_hello():
    print("Hello!")

say_hello()

带参数的装饰器 ​

python
def repeat(n_times):
    def decorator(func):
        def wrapper(*args, **kwargs):
            for _ in range(n_times):
                result = func(*args, **kwargs)
            return result
        return wrapper
    return decorator

@repeat(n_times=3)
def greet(name):
    print(f"Hello, {name}!")

greet("Alice")

类装饰器 ​

python
class CountCalls:
    def __init__(self, func):
        self.func = func
        self.calls = 0
        
    def __call__(self, *args, **kwargs):
        self.calls += 1
        print(f"Call {self.calls} of {self.func.__name__}")
        return self.func(*args, **kwargs)

@CountCalls
def say_hello():
    print("Hello!")

say_hello()  # 输出: Call 1 of say_hello
say_hello()  # 输出: Call 2 of say_hello

上下文管理器 ​

with语句实现 ​

python
# 文件自动关闭
with open('file.txt', 'r') as f:
    content = f.read()

# 自定义上下文管理器
class ManagedFile:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
        
    def __enter__(self):
        self.file = open(self.filename, self.mode)
        return self.file
        
    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.file:
            self.file.close()

# 使用自定义管理器
with ManagedFile('file.txt', 'w') as f:
    f.write("Hello, context!")

contextlib简化实现 ​

python
from contextlib import contextmanager

@contextmanager
def managed_file(filename, mode):
    try:
        f = open(filename, mode)
        yield f
    finally:
        f.close()

# 使用
with managed_file('file.txt', 'w') as f:
    f.write("Using contextlib")

生成器与协程 ​

生成器基础 ​

python
def countdown(n):
    print("Starting countdown")
    while n > 0:
        yield n
        n -= 1
    print("Countdown complete")

# 使用
counter = countdown(5)
for num in counter:
    print(num)

async/await ​

python
import asyncio

async def fetch_data():
    print("Start fetching")
    await asyncio.sleep(2)  # 模拟IO操作
    print("Done fetching")
    return {"data": 123}

async def main():
    task1 = asyncio.create_task(fetch_data())

    await task1

asyncio.run(main())

元类 ​

python
class Meta(type):
    def __new__(cls, name, bases, dct):
        # 添加新属性
        dct['created_by_meta'] = True
        return super().__new__(cls, name, bases, dct)

class MyClass(metaclass=Meta):
    pass

print(MyClass.created_by_meta)  # True

属性拦截 ​

python
class AttributeInterceptor:
    def __getattr__(self, name):
        print(f"Accessing undefined attribute: {name}")
        return None
    
    def __setattr__(self, name, value):
        print(f"Setting attribute: {name} = {value}")
        super().__setattr__(name, value)

obj = AttributeInterceptor()
obj.new_property = 42  # 输出: Setting attribute: new_property = 42
print(obj.undefined)   # 输出: Accessing undefined attribute: undefined

泛型 ​

python
from typing import TypeVar, List, Generic

T = TypeVar('T')

class Stack(Generic[T]):
    def __init__(self) -> None:
        self.items: List[T] = []
    
    def push(self, item: T) -> None:
        self.items.append(item)
    
    def pop(self) -> T:
        return self.items.pop()

# 使用
int_stack = Stack[int]()
int_stack.push(42)

类型联合 ​

python
from typing import Union

def process(value: Union[int, str]) -> None:
    if isinstance(value, int):
        print(f"Integer: {value}")
    else:
        print(f"String: {value}")

process(10)   # Integer: 10
process("Hi") # String: Hi

import aiohttp
import asyncio

async def fetch(url):
    async with aiohttp.ClientSession() as session:
        async with session.get(url) as response:
            return await response.text()

async def main():
    html = await fetch('http://example.com')
    print(html[:100])

asyncio.run(main())

描述符协议 ​

属性描述符 ​

python
class PositiveNumber:
    def __set_name__(self, owner, name):
        self.name = name
    
    def __get__(self, instance, owner):
        return instance.__dict__[self.name]
    
    def __set__(self, instance, value):
        if value <= 0:
            raise ValueError("Value must be positive")
        instance.__dict__[self.name] = value

class Order:
    quantity = PositiveNumber()
    price = PositiveNumber()
    
    def __init__(self, quantity, price):
        self.quantity = quantity
        self.price = price

# 使用
order = Order(10, 25.5)
try:
    order.quantity = -5  # 抛出 ValueError
except ValueError as e:
    print(e)

数据模型特殊方法 ​

运算符重载 ​

python
class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __add__(self, other):
        return Vector(self.x + other.x, self.y + other.y)
    
    def __mul__(self, scalar):
        return Vector(self.x * scalar, self.y * scalar)
    
    def __repr__(self):
        return f"Vector({self.x}, {self.y})"

v1 = Vector(2, 4)
v2 = Vector(1, 3)
print(v1 + v2)  # Vector(3, 7)
print(v1 * 3)   # Vector(6, 12)

上下文管理 ​

python
class Timer:
    def __enter__(self):
        import time
        self.start = time.time()
        return self
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        import time
        self.end = time.time()
        print(f"Elapsed time: {self.end - self.start:.2f} seconds")

# 使用
with Timer():
    # 执行耗时操作
    import time
    time.sleep(2)

自定义容器 ​

python
class CustomList:
    def __init__(self, items):
        self.items = list(items)
    
    def __getitem__(self, index):
        return self.items[index]
    
    def __setitem__(self, index, value):
        self.items[index] = value
    
    def __len__(self):
        return len(self.items)
    
    def __contains__(self, item):
        return item in self.items
    
    def __iter__(self):
        return iter(self.items)

my_list = CustomList([1, 2, 3])
print(len(my_list))  # 3
print(2 in my_list)  # True
for item in my_list:
    print(item)

最佳实践建议 ​

  1. 装饰器适用场景：

    • 添加日志记录
    • 权限验证
    • 性能监控
    • 缓存结果
    • 错误处理

  2. 生成器使用原则：

    • 处理大型数据集时优先使用生成器表达式
    • 流式处理数据时使用生成器函数
    • 需要维护状态时考虑协程

  3. 类型注解指南：

    • 公共API必须添加类型注解
    • 复杂数据结构使用TypedDict
    • 回调函数明确参数和返回类型
    • 定期运行mypy进行类型检查

  4. 异步编程注意事项：

    • 避免在异步函数中使用阻塞IO
    • 使用asyncio.gather并发执行任务
    • 设置合理的超时时间
    • 使用异步上下文管理器管理资源

  5. 元编程使用场景：

    • 框架开发
    • ORM实现
    • 动态API创建
    • 代码生成工具
