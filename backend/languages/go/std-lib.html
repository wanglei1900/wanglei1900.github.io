输入/输出处理 ​

go
import (
    "io"
    "os"
    "bufio"
    "fmt"
)

// 文件读写示例
func fileIOExample() {
    // 创建文件
    file, err := os.Create("test.txt")
    if err != nil {
        panic(err)
    }
    defer file.Close()
    
    // 写入内容
    writer := bufio.NewWriter(file)
    writer.WriteString("Hello, Go Standard Library!\n")
    writer.Flush()
    
    // 读取内容
    reader := bufio.NewReader(file)
    content, _ := io.ReadAll(reader)
    fmt.Println(string(content))
}

网络编程 ​

go
import (
    "net"
    "net/http"
)

// HTTP服务器示例
func httpServerExample() {
    http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
        w.Write([]byte("Hello from Go HTTP Server"))
    })
    
    go http.ListenAndServe(":8080", nil)
}

// TCP客户端示例
func tcpClientExample() {
    conn, err := net.Dial("tcp", "localhost:8080")
    if err != nil {
        panic(err)
    }
    defer conn.Close()
    
    conn.Write([]byte("GET / HTTP/1.0\r\n\r\n"))
    response, _ := io.ReadAll(conn)
    fmt.Println(string(response))
}

并发处理 ​

go
import (
    "sync"
    "time"
)

func concurrencyExample() {
    var wg sync.WaitGroup
    wg.Add(2)
    
    // Goroutine 1
    go func() {
        defer wg.Done()
        time.Sleep(1 * time.Second)
        fmt.Println("Goroutine 1 completed")
    }()
    
    // Goroutine 2
    go func() {
        defer wg.Done()
        time.Sleep(500 * time.Millisecond)
        fmt.Println("Goroutine 2 completed")
    }()
    
    wg.Wait()
    fmt.Println("All goroutines completed")
}

sync - 同步原语 ​

go
var (
    counter int
    mu      sync.Mutex
    once    sync.Once
)

func increment() {
    mu.Lock()
    counter++
    mu.Unlock()
}

func initialize() {
    once.Do(func() {
        fmt.Println("Initialized only once")
    })
}

context - 上下文管理 ​

go
func process(ctx context.Context) {
    select {
    case <-ctx.Done():
        fmt.Println("Operation canceled")
        return
    case <-time.After(5 * time.Second):
        fmt.Println("Operation completed")
    }
}

// 使用示例
ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
defer cancel()
go process(ctx)

testing - 单元测试 ​

go
// math_test.go
package math

import "testing"

func TestAdd(t *testing.T) {
    tests := []struct {
        a, b, expected int
    }{
        {1, 2, 3},
        {-1, 1, 0},
        {0, 0, 0},
    }

    for _, test := range tests {
        result := Add(test.a, test.b)
        if result != test.expected {
            t.Errorf("Add(%d, %d) = %d, expected %d", 
                     test.a, test.b, result, test.expected)
        }
    }
}

// 基准测试
func BenchmarkAdd(b *testing.B) {
    for i := 0; i < b.N; i++ {
        Add(10, 20)
    }
}

最佳实践与技巧 ​

  1. 错误处理

    go
    file, err := os.Open("file.txt")
if err != nil {
    // 处理错误，不要忽略
    log.Printf("Failed to open file: %v", err)
    return
}
defer file.Close()

  2. 资源管理

    • 使用 defer 确保资源释放
    • 对文件、网络连接等资源及时关闭

  3. 高效I/O

    • 使用 bufio 进行缓冲I/O操作
    • 对大文件使用流式处理

  4. 并发安全

    • 使用互斥锁保护共享资源
    • 优先使用通道进行goroutine间通信
    • 使用 context 管理goroutine生命周期

  5. 性能优化

    • 避免不必要的内存分配
    • 使用 sync.Pool 重用对象
    • 使用 pprof 进行性能分析

重要更新 (Go 1.16+) ​

  1. io/fs 包

    go
    import "io/fs"

// 文件系统抽象
var fsys fs.FS = os.DirFS(".")
content, _ := fs.ReadFile(fsys, "file.txt")

  2. embed 包

    go
    import "embed"

// 嵌入静态资源
//go:embed static/*
var staticFiles embed.FS

// 在HTTP服务器中使用
http.Handle("/static/", http.FileServer(http.FS(staticFiles)))

  3. io/ioutil 弃用

    • 使用 os.ReadFile 替代 ioutil.ReadFile
    • 使用 os.WriteFile 替代 ioutil.WriteFile
