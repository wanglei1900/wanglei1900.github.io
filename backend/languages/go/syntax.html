条件语句 ​

go
// if-else
if x > 10 {
    // ...
} else if x > 5 {
    // ...
} else {
    // ...
}

// switch
switch day {
case "Mon":
    fmt.Println("Monday")
case "Tue":
    fmt.Println("Tuesday")
default:
    fmt.Println("Other day")
}

// type switch
switch v := i.(type) {
case int:
    fmt.Printf("Integer: %v\n", v)
case string:
    fmt.Printf("String: %v\n", v)
default:
    fmt.Printf("Unknown type\n")
}

go

    fmt.Println(i)

// while替代
sum := 1
for sum < 1000 {
    sum += sum
}

// 无限循环
for {
    // ...
}

// range循环
arr := []int{1, 2, 3}
for index, value := range arr {
    fmt.Printf("Index: %d, Value: %d\n", index, value)
}

函数 ​

go
// 基本函数
func add(a int, b int) int {
    return a + b
}

// 多返回值
func swap(x, y string) (string, string) {
    return y, x
}

// 命名返回值
func split(sum int) (x, y int) {
    x = sum * 4 / 9
    y = sum - x
    return // 自动返回x,y
}

// 变参函数
func sum(nums ...int) int {
    total := 0
    for _, num := range nums {
        total += num
    }
    return total
}

// 匿名函数
func() {
    fmt.Println("Anonymous function")
}()

结构体与方法 ​

go
// 定义结构体
type Person struct {
    Name string
    Age  int
}

// 实例化
p := Person{"Alice", 30}
p := Person{Name: "Bob", Age: 25}

// 方法定义
func (p Person) SayHello() {
    fmt.Printf("Hello, I'm %s\n", p.Name)
}

// 指针接收者方法
func (p *Person) Birthday() {
    p.Age++
}

接口 ​

go
// 接口定义
type Shape interface {
    Area() float64
    Perimeter() float64
}

// 实现接口
type Rectangle struct {
    Width, Height float64
}

func (r Rectangle) Area() float64 {
    return r.Width * r.Height
}

func (r Rectangle) Perimeter() float64 {
    return 2*(r.Width + r.Height)
}

// 空接口
var any interface{}
any = 42
any = "hello"

错误处理 ​

go
// 错误接口
type error interface {
    Error() string
}

// 创建错误
err := errors.New("something went wrong")

// 错误处理
result, err := someFunction()
if err != nil {
    // 处理错误
    log.Fatal(err)
}

// 自定义错误
type MyError struct {
    Msg string
    Code int
}

func (e *MyError) Error() string {
    return fmt.Sprintf("Error %d: %s", e.Code, e.Msg)
}

Goroutine ​

// 启动goroutine

Channel ​

go
// 创建channel
ch := make(chan int)

// 发送数据
ch <- 42

// 接收数据
value := <-ch

// 带缓冲channel
bufCh := make(chan int, 10)

// 关闭channel
close(ch)

// select语句
select {
case msg := <-ch1:
    fmt.Println("Received", msg)
case ch2 <- 42:
    fmt.Println("Sent 42")
case <-time.After(1 * time.Second):
    fmt.Println("Timeout")
}

同步原语 ​

go
// WaitGroup
var wg sync.WaitGroup
wg.Add(1)
go func() {
    defer wg.Done()
    // ...
}()
wg.Wait()

// Mutex
var mu sync.Mutex
var counter int
mu.Lock()
counter++
mu.Unlock()

最佳实践提示:

  1. 使用go fmt保持代码格式统一
  2. 避免全局变量，优先使用局部变量
  3. 错误处理要明确，不要忽略错误
  4. 并发访问共享资源时使用同步机制
  5. 使用接口提高代码灵活性和可测试性
