python
# 严格的URI验证实现
class RedirectURIValidator:
    def __init__(self, registered_uris):
        self.registered_uris = registered_uris
    
    def validate_redirect_uri(self, requested_uri):
        # 精确匹配，避免子路径漏洞
        if requested_uri not in self.registered_uris:
            raise SecurityError("Invalid redirect URI")
        
        # 检查协议和域名的完整性
        parsed = urlparse(requested_uri)
        if parsed.scheme != 'https' and not parsed.hostname.startswith('localhost'):
            raise SecurityError("Redirect URI must use HTTPS")
        
        return True

JWT客户端认证示例 ​

public class JWTClientAuthentication {
    public String createClientAssertion(String clientId, String tokenEndpoint) {
        long now = System.currentTimeMillis() / 1000L;
        
        JWTClaimsSet claims = new JWTClaimsSet.Builder()
            .issuer(clientId)
            .subject(clientId)
            .audience(tokenEndpoint)
            .expirationTime(new Date(now + 300 * 1000)) // 5分钟
            .issueTime(new Date(now))
            .jwtID(UUID.randomUUID().toString())
            .build();
        
        SignedJWT signedJWT = new SignedJWT(
            new JWSHeader(JWSAlgorithm.RS256),
            claims
        );
        
        // 使用私钥签名
        signedJWT.sign(new RSASSASigner(privateKey));
        return signedJWT.serialize();
    }
}

三、授权端点安全防护 ​

1. State参数防CSRF攻击 ​

javascript
// 生成和验证state参数
class StateParameterManager {
    constructor() {
        this.stateStore = new Map();
    }
    
    generateState(redirectUri, clientId) {
        const state = crypto.randomBytes(32).toString('hex');
        const stateData = {
            value: state,
            redirectUri,
            clientId,
            timestamp: Date.now(),
            // 可添加其他上下文信息
            sessionId: this.getCurrentSessionId()
        };
        
        // 存储state，设置合理过期时间（如10分钟）
        this.stateStore.set(state, stateData);
        setTimeout(() => this.stateStore.delete(state), 10 * 60 * 1000);
        
        return state;
    }
    
    validateState(receivedState, clientId, redirectUri) {
        const stateData = this.stateStore.get(receivedState);
        if (!stateData) {
            throw new Error("Invalid state parameter");
        }
        
        // 清理已使用的state
        this.stateStore.delete(receivedState);
        
        // 验证上下文一致性
        if (stateData.clientId !== clientId || 
            stateData.redirectUri !== redirectUri) {
            throw new Error("State context mismatch");
        }
        
        // 验证会话一致性
        if (stateData.sessionId !== this.getCurrentSessionId()) {
            throw new Error("Session mismatch");
        }
        
        return true;
    }
}

2. PKCE（Proof Key for Code Exchange） ​

PKCE流程实现 ​
python
import hashlib
import base64
import secrets

class PKCEManager:
    @staticmethod
    def generate_code_verifier():
        """生成43-128字符的code_verifier"""
        return secrets.token_urlsafe(96)  # 96字节 -> 128字符
    
    @staticmethod
    def generate_code_challenge(verifier, method='S256'):
        """生成code_challenge"""
        if method == 'S256':
            digest = hashlib.sha256(verifier.encode()).digest()
            return base64.urlsafe_b64encode(digest).decode().replace('=', '')
        elif method == 'plain':
            return verifier
        else:
            raise ValueError("Unsupported challenge method")
    
    def create_pkce_parameters(self):
        verifier = self.generate_code_verifier()
        challenge = self.generate_code_challenge(verifier)
        
        return {
            'code_verifier': verifier,
            'code_challenge': challenge,
            'code_challenge_method': 'S256'
        }

# 在授权请求中包含PKCE参数
authorization_url = (
    f"{authorization_endpoint}?"
    f"response_type=code&"
    f"client_id={client_id}&"
    f"redirect_uri={redirect_uri}&"
    f"code_challenge={challenge}&"
    f"code_challenge_method=S256&"
    f"state={state}"

1. 令牌生命周期管理 ​

java
public class TokenSecurityConfig {
    // 合理的令牌过期时间配置
    private static final long ACCESS_TOKEN_EXPIRY = 3600; // 1小时
    private static final long REFRESH_TOKEN_EXPIRY = 86400 * 30; // 30天
    private static final long AUTHORIZATION_CODE_EXPIRY = 600; // 10分钟
    
    // 令牌格式建议
    public String generateAccessToken(User user, Client client, Set<String> scopes) {
        JWTClaimsSet claims = new JWTClaimsSet.Builder()
            .issuer("https://auth.server.com")
            .subject(user.getId())
            .audience(client.getClientId())
            .expirationTime(new Date(System.currentTimeMillis() + ACCESS_TOKEN_EXPIRY * 1000))
            .issueTime(new Date())
            .claim("scope", String.join(" ", scopes))
            .claim("client_id", client.getClientId())
            .claim("token_type", "Bearer")
            // 添加令牌唯一标识
            .jwtID(generateTokenId())
            .build();
        
        return signJWT(claims);
    }
}

推荐的令牌存储实现 ​
javascript
class TokenStorage {
    constructor() {
        this.accessToken = null;
        this.refreshToken = null;
        this.tokenExpiry = null;
    }
    
    // 内存存储访问令牌
    setTokens(accessToken, refreshToken, expiresIn) {
        this.accessToken = accessToken;
        this.refreshToken = refreshToken;
        this.tokenExpiry = Date.now() + (expiresIn * 1000);
        
        // 刷新令牌可安全存储在HttpOnly cookie中
        document.cookie = `refresh_token=${refreshToken}; HttpOnly; Secure; SameSite=Strict; Path=/auth/refresh`;
    }
    
    // 自动令牌刷新
    async getValidAccessToken() {
        if (this.isTokenExpired() && this.refreshToken) {
            await this.refreshAccessToken();
        }
        return this.accessToken;
    }
    
    isTokenExpired() {
        return Date.now() >= this.tokenExpiry - 60000; // 提前1分钟刷新
    }
}

3. 令牌撤销与清单 ​

python
class TokenRevocationService:
    def __init__(self, redis_client):
        self.redis = redis_client
    
    def revoke_token(self, token_jti, reason="user_revoked", expires_in=3600):
        """撤销令牌"""
        revocation_key = f"revoked_token:{token_jti}"
        self.redis.setex(revocation_key, expires_in, reason)
    
    def is_token_revoked(self, token_jti):
        """检查令牌是否被撤销"""
        return self.redis.exists(f"revoked_token:{token_jti}")
    
    def bulk_revoke_user_tokens(self, user_id, client_id=None):
        """撤销用户的所有令牌"""
        pattern = f"token:{user_id}:*"
        if client_id:
            pattern = f"token:{user_id}:{client_id}:*"
        
        for key in self.redis.keys(pattern):
            token_jti = key.decode().split(":")[-1]
            self.revoke_token(token_jti, "user_logout")

五、针对常见攻击的防护措施 ​

1. 混合流检测与防护 ​

java
public class AuthorizationEndpoint {
    public AuthorizationResponse processAuthorizationRequest(AuthorizationRequest request) {
        // 检测混合流攻击迹象
        if (isPotentialMixUpAttack(request)) {
            throw new SecurityException("Potential mix-up attack detected");
        }
        
        // 验证response_type组合的合理性
        if (!isValidResponseTypeCombination(request.getResponseType())) {
            throw new SecurityException("Invalid response_type combination");
        }
        
        return createAuthorizationResponse(request);
    }
    
    private boolean isPotentialMixUpAttack(AuthorizationRequest request) {
        // 检查是否尝试同时获取多种令牌类型
        Set<String> responseTypes = request.getResponseTypes();
        
        // 可疑的组合：code + token 或 code + id_token
        if (responseTypes.contains("code") && 
           (responseTypes.contains("token") || responseTypes.contains("id_token"))) {
            logger.warn("Potential mix-up attack detected for client: {}", 
                       request.getClientId());
            return true;
        }
        
        return false;
    }
}

2. 点击劫持防护 ​

python
# 授权页面的安全头部设置
class SecurityHeadersMiddleware:
    def __init__(self, app):
        self.app = app
    
    def __call__(self, environ, start_response):
        def security_headers_start_response(status, headers, exc_info=None):
            # 添加安全头部
            security_headers = [
                ('X-Frame-Options', 'DENY'),  # 防止点击劫持
                ('Content-Security-Policy', "frame-ancestors 'none'"),
                ('X-Content-Type-Options', 'nosniff'),
                ('Referrer-Policy', 'strict-origin-when-cross-origin')
            ]
            
            headers.extend(security_headers)
            return start_response(status, headers, exc_info)
        
        return self.app(environ, security_headers_start_response)

1. 安全事件日志 ​

javascript
class SecurityLogger {
    logAuthorizationEvent(event) {
        const logEntry = {
            timestamp: new Date().toISOString(),
            eventType: event.type,
            clientId: event.clientId,
            userId: event.userId,
            ipAddress: event.ipAddress,
            userAgent: event.userAgent,
            scope: event.scope,
            riskLevel: this.calculateRiskLevel(event),
            // 不记录敏感信息如令牌、密码等
            metadata: {
                redirectUri: event.redirectUri,
                responseType: event.responseType
            }
        };
        
        // 发送到安全信息事件管理系统
        this.sendToSIEM(logEntry);
        
        // 高风险事件实时告警
        if (logEntry.riskLevel === 'high') {
            this.triggerAlert(logEntry);
        }
    }
    
    calculateRiskLevel(event) {
        let riskScore = 0;
        
        // 基于多种因素计算风险分数
        if (event.ipAddress && this.isSuspiciousIP(event.ipAddress)) riskScore += 30;
        if (event.userAgent && this.isSuspiciousUA(event.userAgent)) riskScore += 20;
        if (event.scope && this.hasHighPrivilegeScopes(event.scope)) riskScore += 25;
        if (this.isUnusualTime(event)) riskScore += 15;
        
        if (riskScore >= 60) return 'high';
        if (riskScore >= 30) return 'medium';
        return 'low';
    }
}

2. 实时威胁检测 ​

python
class ThreatDetection:
    def __init__(self):
        self.failed_attempts = defaultdict(int)
        self.suspicious_clients = set()
    
    def detect_anomalies(self, auth_request):
        client_id = auth_request.client_id
        user_id = auth_request.user_id
        
        # 检测暴力破解
        if self.is_brute_force_attempt(client_id, user_id):
            self.block_client_temporarily(client_id)
            return True
        
        # 检测地理异常
        if self.has_geographic_anomaly(auth_request):
            self.require_additional_authentication(auth_request)
            return True
        
        # 检测行为异常
        if self.has_behavioral_anomaly(auth_request):
            self.flag_for_review(auth_request)
            return True
        
        return False
    
    def is_brute_force_attempt(self, client_id, user_id):
        key = f"{client_id}:{user_id}"
        self.failed_attempts[key] += 1
        
        # 5分钟内失败5次以上视为暴力破解
        return self.failed_attempts[key] > 5
