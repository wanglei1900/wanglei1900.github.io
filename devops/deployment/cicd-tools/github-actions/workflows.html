name: CI/CD Pipeline                    # 工作流名称
on:                                     # 触发条件
  push:
    branches: [ main, develop ]
    paths:
      - 'src/**'
      - 'package.json'
      - '.github/workflows/**'
  
  pull_request:
    branches: [ main ]
    types: [opened, synchronize, reopened]
  
  schedule:
    - cron: '0 2 * * 1-5'              # 工作日凌晨2点
  
  workflow_dispatch:                    # 手动触发
    inputs:
      environment:
        description: '部署环境'
        required: true
        default: 'staging'
        type: choice
        options:
          - staging
          - production
      
      version:
        description: '部署版本'
        required: false
        type: string

env:                                    # 全局环境变量
  NODE_VERSION: '18.x'
  REGISTRY: 'ghcr.io'
  DOCKER_IMAGE_NAME: '${{ github.repository }}'

concurrency:                           # 并发控制
  group: ${{ github.ref }}
  cancel-in-progress: true

jobs:                                  # 作业定义开始
  test:
    name: Run Tests
    runs-on: ubuntu-latest
    environment: test

      - main

3. 作业设计与优化策略 ​

3.1 作业依赖与条件控制 ​

yaml
jobs:
  lint:
    name: Lint Code
    runs-on: ubuntu-latest
    outputs:                          # 作业输出定义
      eslint_status: ${{ steps.eslint.outcome }}
      passed: ${{ steps.check.outputs.passed }}
    steps:
      - name: Run ESLint
        id: eslint
        run: npm run lint
        
      - name: Check results
        id: check
        run: |
          if [ $? -eq 0 ]; then
            echo "passed=true" >> $GITHUB_OUTPUT
          else
            echo "passed=false" >> $GITHUB_OUTPUT
          fi

  test:
    name: Run Tests
    runs-on: ubuntu-latest
    needs: lint                       # 依赖lint作业
    if: |
      needs.lint.outputs.passed == 'true' &&
      github.event_name == 'pull_request'
    strategy:
      matrix:
        node-version: [16.x, 18.x]
        os: [ubuntu-latest, windows-latest]
    steps:
      - uses: actions/checkout@v4
      
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node-version }}
          cache: 'npm'
      
      - name: Install dependencies
        run: npm ci
      
      - name: Run tests
        run: npm test
        env:
          CI: true
          NODE_ENV: test
      
      - name: Upload coverage
        uses: codecov/codecov-action@v3
        with:
          token: ${{ secrets.CODECOV_TOKEN }}

  deploy:
    name: Deploy to Production
    runs-on: ubuntu-latest
    needs: [lint, test]              # 依赖多个作业
    if: |
      always() &&                    # 即使前面作业失败也执行
      github.ref == 'refs/heads/main' &&
      contains(needs.*.result, 'success')
    environment: production
    permissions:
      contents: read
      deployments: write
    steps:
      # 部署步骤

3.2 矩阵策略高级用法 ​

yaml
jobs:
  build-and-test:
    runs-on: ${{ matrix.os }}
    strategy:
      matrix:
        # 基础矩阵
        os: [ubuntu-latest, windows-latest, macos-latest]
        node-version: [16.x, 18.x, 20.x]
        
        # 排除特定组合
        exclude:
          - os: windows-latest
            node-version: 20.x
        
        # 包含额外组合
        include:
          - os: ubuntu-latest
            node-version: 14.x
            experimental: true
          
          - os: ubuntu-latest
            node-version: 22.x
            experimental: true
        
        # 失败处理策略
        fail-fast: false              # 一个失败不停止其他
        
        # 最大并行数
        max-parallel: 4
    
    name: Build on ${{ matrix.os }} with Node ${{ matrix.node-version }}
    steps:
      - name: Checkout
        uses: actions/checkout@v4
      
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node-version }}
      
      - name: Build
        run: npm run build
      
      - name: Test
        if: matrix.experimental != 'true'
        run: npm test

4. 步骤(Step)高级技巧 ​

4.1 步骤条件与错误处理 ​

yaml
steps:
  - name: Check code style
    id: style-check
    continue-on-error: true          # 失败继续执行
    run: npm run lint
    
  - name: Analyze lint results
    if: steps.style-check.outcome == 'failure'
    run: |
      echo "Linting failed but continuing..."
      # 生成报告
      npm run lint:report > lint-report.json
      
  - name: Upload lint report
    if: always() && steps.style-check.outcome == 'failure'
    uses: actions/upload-artifact@v3
    with:
      name: lint-report
      path: lint-report.json
  
  - name: Run tests
    timeout-minutes: 30              # 设置超时
    run: npm test
  
  - name: Handle test failure
    if: failure()                    # 仅在失败时执行
    run: |
      echo "Tests failed! Sending notification..."
      # 发送通知逻辑
      
  - name: Always run cleanup
    if: always()                     # 总是执行
    run: |
      echo "Cleaning up resources..."
      # 清理逻辑
  
  - name: Conditional step
    if: |
      github.event_name == 'push' &&
      github.ref == 'refs/heads/main' &&
      !contains(github.event.commits[0].message, '[skip-ci]')
    run: echo "This step runs only on main branch push without [skip-ci]"

5. 缓存与构件管理 ​

5.1 智能缓存策略 ​

yaml
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
      
      - name: Cache Node.js modules
        id: cache-npm
        uses: actions/cache@v3
        with:
          path: |
            ~/.npm
            node_modules
            **/node_modules
          key: ${{ runner.os }}-node-${{ hashFiles('**/package-lock.json') }}
          restore-keys: |
            ${{ runner.os }}-node-
            ${{ runner.os }}-
      
      - name: Cache Docker layers
        uses: actions/cache@v3
        with:
          path: /tmp/.buildx-cache
          key: ${{ runner.os }}-buildx-${{ github.sha }}
          restore-keys: |
            ${{ runner.os }}-buildx-
      
      - name: Cache pip packages
        uses: actions/cache@v3
        with:
          path: ~/.cache/pip
          key: ${{ runner.os }}-pip-${{ hashFiles('**/requirements.txt') }}
      
      - name: Install dependencies

    runs-on: ubuntu-latest
    needs: build-image
    environment: ${{ github.ref == 'refs/heads/main' && 'production' || 'staging' }}
    strategy:
      matrix:
        environment: ['staging', 'production']
        include:
          - environment: 'staging'
            if: github.ref == 'refs/heads/develop'
            namespace: 'staging'
            k8s_file: 'k8s/staging'
          
          - environment: 'production'
            if: github.ref == 'refs/heads/main'
            namespace: 'production'
            k8s_file: 'k8s/production'
    
    if: ${{ matrix.if }}
    steps:
      - uses: actions/checkout@v4
      
      - name: Configure Kubernetes
        uses: azure/setup-kubectl@v3
      
      - name: Deploy to ${{ matrix.environment }}
        run: |
          kubectl apply -f ${{ matrix.k8s_file }} --namespace ${{ matrix.namespace }}
          kubectl rollout status deployment/app --namespace ${{ matrix.namespace }}
      
      - name: Run smoke tests
        run: |
          npm run test:smoke -- --environment=${{ matrix.environment }}
      
      - name: Notify deployment
        uses: 8398a7/action-slack@v3
        with:
          status: ${{ job.status }}
          channel: '#deployments'
        env:
          SLACK_WEBHOOK_URL: ${{ secrets.SLACK_WEBHOOK_URL }}

  # 阶段6: 监控与清理
  post-deployment:
    name: Post-Deployment Tasks
    runs-on: ubuntu-latest
    needs: deploy
    if: always()
    steps:
      - name: Clean up old deployments
        run: |
          # 清理旧的Kubernetes资源
          kubectl delete pods --field-selector=status.phase=Succeeded --namespace=default
      
      - name: Update deployment tracker
        uses: actions/github-script@v6
        with:
          script: |
            await github.rest.issues.create({
              owner: context.repo.owner,
              repo: context.repo.repo,
              title: `Deployment completed: ${context.sha}`,
              body: `Deployment of ${context.sha} completed at ${new Date().toISOString()}`
            })
      
      - name: Generate deployment report
        run: |
          cat << EOF > deployment-report.md
          # Deployment Report
          
          ## Summary
          - **Status**: ${{ needs.deploy.result }}
          - **Environment**: ${{ needs.deploy.outputs.environment }}
          - **Image**: ${{ needs.build-image.outputs.image_tag }}
          - **Timestamp**: $(date)
          
          ## Rollback Information
          To rollback, run:
          \`\`\`
          kubectl rollout undo deployment/app --namespace=${{ needs.deploy.outputs.namespace }}
          \`\`\`
          EOF
      
      - name: Upload deployment report
        uses: actions/upload-artifact@v3
        with:
          name: deployment-report-${{ github.sha }}
          path: deployment-report.md

10. 最佳实践总结 ​

10.1 设计原则 ​

  1. 保持工作流模块化：使用可重用工作流和自定义Action
  2. 实施渐进式部署：分阶段验证和部署
  3. 优化执行性能：利用缓存、并行执行和选择性触发
  4. 确保安全性：最小权限原则和秘密管理
  5. 完善监控和调试：全面的日志和指标收集
