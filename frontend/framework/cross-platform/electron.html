main.js

javascript
const { app, BrowserWindow } = require('electron')

function createWindow() {
  const win = new BrowserWindow({
    width: 800,
    height: 600,
    webPreferences: {
      nodeIntegration: true,
      contextIsolation: false
    }
  })

  win.loadFile('index.html')

app.whenReady().then(createWindow)

app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') app.quit()
})

进程间通信 (IPC) ​

基础 IPC 实现 ​

渲染进程 (renderer.js)

javascript
const { ipcRenderer } = require('electron')

// 发送消息到主进程
document.getElementById('btn').addEventListener('click', () => {
  ipcRenderer.send('message-from-renderer', 'Hello Main Process!')
})

// 接收主进程回复
ipcRenderer.on('reply-to-renderer', (event, arg) => {
  console.log('Main replied:', arg)
})

主进程 (main.js)

javascript
const { ipcMain } = require('electron')

ipcMain.on('message-from-renderer', (event, arg) => {
  console.log('Renderer says:', arg)
  
  // 回复渲染进程
  event.sender.send('reply-to-renderer', 'Hello from Main Process!')
})

高级通信模式 ​

javascript
// 双向异步通信
ipcRenderer.invoke('perform-action', data)
  .then(result => console.log(result))

ipcMain.handle('perform-action', async (event, data) => {
  const result = await someAsyncOperation(data)
  return result
})

文件系统操作 ​

javascript
const fs = require('fs')
const path = require('path')
const { dialog } = require('electron')

// 保存文件对话框
async function saveFile(content) {
  const { filePath } = await dialog.showSaveDialog({
    title: '保存文件',
    filters: [{ name: '文本文件', extensions: ['txt'] }]
  })
  
  if (filePath) {
    fs.writeFileSync(filePath, content)
    return true
  }
  return false
}

系统托盘实现 ​

javascript
const { Tray, Menu } = require('electron')
const path = require('path')

let tray = null

app.whenReady().then(() => {
  tray = new Tray(path.join(__dirname, 'icon.png'))
  
  const contextMenu = Menu.buildFromTemplate([
    { label: '显示', click: () => mainWindow.show() },
    { label: '退出', click: () => app.quit() }
  ])
  
  tray.setToolTip('我的 Electron 应用')
  tray.setContextMenu(contextMenu)
  
  tray.on('click', () => mainWindow.show())
})

自动更新实现 ​

javascript
const { autoUpdater } = require('electron-updater')

function setupAutoUpdate() {
  autoUpdater.autoDownload = true
  autoUpdater.autoInstallOnAppQuit = true
  
  autoUpdater.on('update-available', () => {
    mainWindow.webContents.send('update-status', '下载更新中...')
  })
  
  autoUpdater.on('update-downloaded', () => {
    mainWindow.webContents.send('update-status', '准备安装更新')
    autoUpdater.quitAndInstall()
  })
  
  autoUpdater.checkForUpdates()
}

安全配置示例 ​

javascript
new BrowserWindow({
  webPreferences: {
    // 启用上下文隔离

    // 禁用Node.js集成
    nodeIntegration: false,

// 禁用危险功能
win.webContents.session.setPermissionRequestHandler((webContents, permission, callback) => {
  const allowedPermissions = ['notifications']
  callback(allowedPermissions.includes(permission))
})

代码组织原则 ​

  1. 分离主进程和渲染进程代码
  2. 使用IPC进行进程通信
  3. 将业务逻辑与UI分离
  4. 对敏感操作进行额外验证
  5. 实现错误边界和全局错误处理

错误处理 ​

javascript
// 主进程全局错误处理
process.on('uncaughtException', (error) => {
  console.error('未捕获异常:', error)
  dialog.showErrorBox('应用错误', '发生未预期错误')
})

// 渲染进程错误处理
window.addEventListener('error', (event) => {
  ipcRenderer.send('renderer-error', {
    message: event.error.message,
    stack: event.error.stack
  })
})

// IPC错误处理
ipcMain.handle('safe-operation', async (event, ...args) => {
  try {
    return await performOperation(...args)
  } catch (error) {
    console.error('操作失败:', error)
    throw new Error('操作失败，请重试')
  }
})

常见问题解决方案 ​

1. 白屏问题 ​

原因：渲染进程加载失败
解决：

javascript
win.webContents.on('did-fail-load', (event, errorCode, errorDescription) => {
  console.error(`加载失败: ${errorCode} ${errorDescription}`)
  win.loadFile('fallback.html')
})

2. 应用菜单不显示 ​

原因：macOS 特殊处理
解决：

javascript
app.whenReady().then(() => {
  createWindow()
  
  // macOS 特殊处理
  app.on('activate', () => {
    if (BrowserWindow.getAllWindows().length === 0) createWindow()
  })
})

3. 打包后资源路径错误 ​

解决：使用 app.getAppPath()

javascript
const path = require('path')
const appPath = app.getAppPath()

const iconPath = path.join(appPath, 'assets', 'icon.png')

const { exec } = require('child_process')

// 执行系统命令
function runCommand(cmd, callback) {
  exec(cmd, (error, stdout, stderr) => {
    if (error) {
      console.error(`执行错误: ${error}`)
      return callback(error)
    }

  })
}
