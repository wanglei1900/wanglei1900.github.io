1. 资源预加载概述 ​

什么是资源预加载？ ​

资源预加载（Resource Preloading）是指在浏览器实际需要某些资源（如图片、脚本、样式表、字体等）之前，提前开始加载这些资源的技术。通过合理使用预加载，可以显著减少关键资源的加载时间，从而提升页面性能指标（如LCP、FCP等）。

预加载的核心价值 ​

  • 减少关键资源加载时间：提前加载首屏所需资源
  • 优化用户感知性能：消除渲染阻塞，提升流畅度
  • 提高资源缓存利用率：提前缓存后续页面资源
  • 平衡带宽利用：利用空闲带宽预加载非关键资源

预加载 vs 懒加载 ​

特性预加载懒加载
时机页面加载早期或空闲时资源进入视口时
目标关键资源/后续页面资源非首屏资源
优先级高优先级低优先级
适用场景首屏内容、核心功能资源非首屏图片、次要内容

预加载适用场景 ​

  1. 首屏关键渲染路径资源（CSS、JS）
  2. 首屏大图/英雄区域图片
  3. 自定义字体文件
  4. 后续页面核心资源（单页应用路由）
  5. 关键交互所需的资源

2. 浏览器原生预加载机制 ​

2.1 Preload ​

html
<!-- 基本用法 -->
<link rel="preload" href="font.woff2" as="font" type="font/woff2" crossorigin>

<!-- 媒体查询适配 -->
<link rel="preload" href="large-image.jpg" as="image" media="(min-width: 600px)">

<!-- 脚本预加载 -->
<link rel="preload" href="critical.js" as="script">

特性：

  • 高优先级加载
  • 不阻塞渲染
  • 支持多种资源类型（as属性）
  • 可指定跨域请求（crossorigin）

支持的资源类型：

  • script - JavaScript文件
  • style - CSS样式表
  • image - 图片资源
  • font - 字体文件
  • document - HTML文档
  • fetch - XHR/fetch请求

2.2 Prefetch ​

html
<!-- 预取后续页面资源 -->
<link rel="prefetch" href="next-page.js" as="script">

使用场景：

  • 下一页可能需要的资源
  • 非关键功能模块
  • 用户可能访问的路径资源

2.3 Preconnect/DNS-Prefetch ​

html
<!-- 提前建立连接 -->
<link rel="preconnect" href="https://api.example.com">

<!-- 仅DNS预解析 -->
<link rel="dns-prefetch" href="https://cdn.example.com">

适用场景：

  • 关键第三方资源（Google Fonts、CDN资源）
  • 跨域API请求
  • 社交分享按钮资源

3. JavaScript预加载技术 ​

3.1 图片预加载 ​

javascript
// 简单图片预加载
function preloadImage(url) {
  const img = new Image();
  img.src = url;
}

// 带回调的批量预加载
function preloadImages(urls, callback) {
  let loaded = 0;
  urls.forEach(url => {
    const img = new Image();
    img.onload = () => {
      loaded++;
      if (loaded === urls.length) callback();
    };
    img.src = url;
  });
}

3.2 CSS/JS预加载 ​

javascript
// 动态创建link标签预加载CSS
function preloadCSS(url) {
  const link = document.createElement('link');
  link.rel = 'preload';
  link.href = url;
  link.as = 'style';
  document.head.appendChild(link);
}

// 动态预加载JS
function preloadJS(url) {
  const script = document.createElement('script');
  script.src = url;
  script.async = true;
  document.head.appendChild(script);
}

3.3 Fetch API预加载 ​

javascript
// 预加载数据资源
function preloadData(url) {
  fetch(url, {
    method: 'GET',
    headers: {
      'Content-Type': 'application/json'
    },
    credentials: 'include',
    cache: 'force-cache'
  }).catch(e => console.error('Preload failed', e));
}

// 预加载并缓存
async function preloadAndCache(url) {
  const cache = await caches.open('preload-cache');
  const response = await fetch(url);
  await cache.put(url, response.clone());
  return response;
}

3.4 Service Worker预加载 ​

javascript
// Service Worker安装阶段预缓存
self.addEventListener('install', event => {
  event.waitUntil(
    caches.open('v1').then(cache => {
      return cache.addAll([
        '/critical.css',
        '/main.js',
        '/hero-image.jpg'
      ]);
    })
  );
});

// 运行时预取资源
self.addEventListener('message', event => {
  if (event.data.action === 'prefetch') {
    event.ports[0].postMessage({status: 'started'});
    event.waitUntil(
      caches.open('runtime-cache').then(cache => {
        return cache.addAll(event.data.resources);
      })
    );
  }
});

4.2 基于网络状况的动态预加载 ​

javascript
function adaptivePreload() {
  if (navigator.connection) {
    const connection = navigator.connection;
    
    // 节省流量模式
    if (connection.saveData) {
      return;
    }
    
    // 根据网络类型调整
    switch (connection.effectiveType) {
      case '4g':
        preloadResource('large-image.jpg', 'image');
        preloadResource('optional-module.js', 'script');
        break;
      case '3g':
        preloadResource('medium-image.jpg', 'image');
        break;
      case '2g':
        // 仅预加载关键资源
        preloadResource('critical.css', 'style');
        break;
    }
  } else {
    // 默认策略
    preloadResource('critical.css', 'style');
    preloadResource('main.js', 'script');
  }
}

4.3 基于用户行为的预测预加载 ​

javascript
// 鼠标悬停在链接上时预加载
document.querySelectorAll('a[data-prefetch]').forEach(link => {
  link.addEventListener('mouseenter', () => {
    const target = link.getAttribute('href');
    preloadResource(target, 'document');
  }, { once: true });
});

// 滚动到特定区域预加载
const observer = new IntersectionObserver(entries => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const module = entry.target.dataset.module;
      preloadResource(`${module}.js`, 'script');
      observer.unobserve(entry.target);
    }
  });
}, { rootMargin: '200px 0px' });

document.querySelectorAll('[data-module]').forEach(el => {
  observer.observe(el);
});

4.4 预加载与懒加载的平衡 ​

javascript
// 首屏资源：Preload
preloadResource('hero-image.jpg', 'image');

// 非首屏资源：IntersectionObserver懒加载
const lazyImages = document.querySelectorAll('img.lazy');
const imgObserver = new IntersectionObserver((entries, observer) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const img = entry.target;
      img.src = img.dataset.src;
      img.classList.remove('lazy');
      observer.unobserve(img);
    }
  });
});

lazyImages.forEach(img => imgObserver.observe(img));

5. 框架集成预加载 ​

5.1 React动态导入预加载 ​

jsx
import React, { Suspense, lazy } from 'react';

// 提前开始加载组件
const OtherComponent = lazy(() => import('./OtherComponent'));

// 交互时预加载
function PreloadButton() {
  const handleHover = () => {
    import('./OtherComponent').then(module => {
      // 组件已加载但尚未渲染
    });
  };
  
  return (
    <button onMouseEnter={handleHover}>
      悬停预加载
    </button>
  );
}

function MyComponent() {
  return (
    <Suspense fallback={<div>Loading...</div>}>
      <OtherComponent />
      <PreloadButton />
    </Suspense>
  );
}

5.2 Vue路由预加载 ​

javascript
// vue-router配置
const router = new VueRouter({
  routes: [
    {
      path: '/dashboard',
      component: () => import(/* webpackPrefetch: true */ './Dashboard.vue')
    },
    {
      path: '/settings',
      component: () => import(/* webpackPreload: true */ './Settings.vue')
    }
  ]
});

// 编程式预加载
router.beforeEach((to, from, next) => {
  if (to.meta.preload) {
    const components = router.getMatchedComponents(to);
    components.forEach(component => {
      if (typeof component === 'function') {
        component();
      }
    });
  }
  next();
});

6.2 预加载命中率计算 ​

javascript
function checkPreloadCacheHit(resourceUrl) {
  const entries = performance.getEntriesByName(resourceUrl);
  
  if (entries.length === 0) return false;
  
  const entry = entries[0];
  return entry.transferSize === 0 || entry.decodedBodySize === 0;
}

// 预加载有效性分析
const preloadedResources = [
  'critical.css',
  'main.js',
  'hero-image.jpg'
];

preloadedResources.forEach(url => {
  const hit = checkPreloadCacheHit(url);
  console.log(`${url}: ${hit ? '命中缓存' : '未命中缓存'}`);
});

7.2 案例二：单页应用路由优化 ​

问题： 路由切换延迟明显
解决方案：

javascript
// 路由预加载策略
const routes = [
  { path: '/', component: Home },
  { path: '/about', preload: () => import('./About') },
  { path: '/contact', preload: () => import('./Contact') }
];

// 路由变化时预加载
router.beforeEach((to, from, next) => {
  const route = routes.find(r => r.path === to.path);
  
  if (route && route.preload) {
    route.preload().then(() => next());
  } else {
    next();
  }
});

// 悬停预加载导航项
document.querySelectorAll('.nav-link').forEach(link => {
  link.addEventListener('mouseenter', () => {
    const path = link.getAttribute('href');
    const route = routes.find(r => r.path === path);
    if (route && route.preload) route.preload();
  });
});

结果： 路由切换时间减少65%

8. 常见问题与解决方案 ​

8.1 预加载过多导致带宽竞争 ​

问题： 同时预加载过多资源，影响关键资源加载
解决方案：

javascript
class PreloadQueue {
  constructor(maxConcurrent = 3) {
    this.queue = [];
    this.inProgress = 0;
    this.maxConcurrent = maxConcurrent;
  }

  add(resource, priority = 0) {
    this.queue.push({ resource, priority });
    this.queue.sort((a, b) => b.priority - a.priority);
    this.process();
  }

  process() {
    while (this.inProgress < this.maxConcurrent && this.queue.length) {
      const { resource } = this.queue.shift();
      this.inProgress++;
      
      preloadResource(resource).finally(() => {
        this.inProgress--;
        this.process();
      });
    }
  }
}

// 使用示例
const preloader = new PreloadQueue();
preloader.add('critical.js', 10);
preloader.add('hero-image.jpg', 9);
preloader.add('secondary-image.jpg', 5);

8.2 预加载资源未使用 ​

问题： 预加载了用户未访问的资源
解决方案：

  • 基于用户行为预测（点击、悬停）
  • 设置预加载超时时间
  • 监控实际使用率优化策略
javascript
function preloadWithTimeout(url, as, timeout = 3000) {
  const link = document.createElement('link');
  link.rel = 'preload';
  link.href = url;
  link.as = as;
  
  document.head.appendChild(link);
  
  // 超时后移除
  setTimeout(() => {
    if (link.parentNode) {
      link.parentNode.removeChild(link);
    }
  }, timeout);
}
