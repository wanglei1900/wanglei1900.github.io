3. 卷积滤镜 ​

javascript
function applyConvolutionFilter(ctx, kernel, offset = 0) {
    const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
    const data = imageData.data;
    const tempData = new Uint8ClampedArray(data);
    
    const kSize = Math.sqrt(kernel.length);
    const kRadius = Math.floor(kSize / 2);
    
    for (let y = kRadius; y < canvas.height - kRadius; y++) {
        for (let x = kRadius; x < canvas.width - kRadius; x++) {
            let r = 0, g = 0, b = 0;
            
            for (let ky = 0; ky < kSize; ky++) {
                for (let kx = 0; kx < kSize; kx++) {
                    const idx = ((y + ky - kRadius) * canvas.width + (x + kx - kRadius)) * 4;
                    const k = kernel[ky * kSize + kx];
                    
                    r += tempData[idx] * k;
                    g += tempData[idx + 1] * k;
                    b += tempData[idx + 2] * k;
                }
            }
            
            const idx = (y * canvas.width + x) * 4;
            data[idx] = r + offset;
            data[idx + 1] = g + offset;
            data[idx + 2] = b + offset;
        }
    }
    
    ctx.putImageData(imageData, 0, 0);
}

// 使用边缘检测滤镜
const edgeDetectionKernel = [
    -1, -1, -1,
    -1,  8, -1,
    -1, -1, -1
];

applyConvolutionFilter(ctx, edgeDetectionKernel);

let lastTime = 0;

function animate(timestamp) {
    requestAnimationFrame(animate);
    
    // 计算时间增量
    const deltaTime = timestamp - lastTime;

1. 数据可视化 ​

javascript
function drawBarChart(data) {
    const padding = 40;
    const barWidth = (canvas.width - padding * 2) / data.length;
    const maxValue = Math.max(...data);
    
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    
    // 绘制坐标轴
    ctx.beginPath();
    ctx.moveTo(padding, padding);
    ctx.lineTo(padding, canvas.height - padding);
    ctx.lineTo(canvas.width - padding, canvas.height - padding);
    ctx.stroke();
    
    // 绘制数据条
    data.forEach((value, i) => {
        const barHeight = (value / maxValue) * (canvas.height - padding * 2);
        const x = padding + i * barWidth;
        const y = canvas.height - padding - barHeight;
        
        // 渐变填充
        const gradient = ctx.createLinearGradient(x, y, x, y + barHeight);
        gradient.addColorStop(0, '#4facfe');
        gradient.addColorStop(1, '#00f2fe');
        
        ctx.fillStyle = gradient;
        ctx.fillRect(x + 2, y, barWidth - 4, barHeight);
    });
}

// 亮度调整
function adjustBrightness(percent) {
    const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
    const data = imageData.data;
    const factor = percent / 100;
    
    for (let i = 0; i < data.length; i += 4) {
        data[i] = clamp(data[i] * factor, 0, 255);     // R
        data[i + 1] = clamp(data[i + 1] * factor, 0, 255); // G
        data[i + 2] = clamp(data[i + 2] * factor, 0, 255); // B
    }
    
    ctx.putImageData(imageData, 0, 0);
}

function clamp(value, min, max) {
    return Math.min(Math.max(value, min), max);
}

3. 物理引擎集成 ​

javascript
// 简单物理系统实现
class PhysicsObject {
    constructor(x, y, radius) {
        this.x = x;
        this.y = y;
        this.radius = radius;
        this.vx = 0;
        this.vy = 0;
        this.gravity = 0.5;
        this.friction = 0.99;
        this.elasticity = 0.8;
    }
    
    update() {
        this.vy += this.gravity;
        this.x += this.vx;
        this.y += this.vy;
        this.vx *= this.friction;
        this.vy *= this.friction;
        
        // 边界碰撞检测
        if (this.x + this.radius > canvas.width) {
            this.x = canvas.width - this.radius;
            this.vx = -this.vx * this.elasticity;
        } else if (this.x - this.radius < 0) {
            this.x = this.radius;
            this.vx = -this.vx * this.elasticity;
        }
        
        if (this.y + this.radius > canvas.height) {
            this.y = canvas.height - this.radius;
            this.vy = -this.vy * this.elasticity;
        } else if (this.y - this.radius < 0) {
            this.y = this.radius;
            this.vy = -this.vy * this.elasticity;
        }
    }
    
    draw(ctx) {
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
        ctx.fillStyle = '#FF5733';
        ctx.fill();
        ctx.stroke();
    }
}

const worker = new Worker('canvas-worker.js');

const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);

// canvas-worker.js
self.onmessage = function(e) {
    if (e.data.cmd === 'process') {
        const imageData = e.data.imageData;
        const data = imageData.data;
        
        // 在Worker中处理像素
        for (let i = 0; i < data.length; i += 4) {
            // 例如应用灰度滤镜
            const avg = (data[i] + data[i + 1] + data[i + 2]) / 3;
            data[i] = avg;     // R
            data[i + 1] = avg; // G
            data[i + 2] = avg; // B
        }
        
        self.postMessage({ imageData: imageData }, [imageData.data.buffer]);
    }

    frameCount++;

        const fps = Math.round((frameCount * 1000) / (timestamp - lastFpsUpdate));
