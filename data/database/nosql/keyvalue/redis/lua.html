1.1 Lua脚本概述 ​

1.1.1 为什么需要Lua脚本？ ​

  1. 原子性操作：复杂操作需要原子性执行
  2. 减少网络开销：多个命令合并为一个脚本执行
  3. 性能优化：在服务端执行，避免多次往返
  4. 逻辑封装：封装复杂业务逻辑

  • Redis 2.6+：支持Lua脚本

三、Lua脚本高级特性 ​

3.1 原子性与事务 ​

3.1.1 原子性保证 ​

lua
-- Lua脚本中的操作是原子的
local function atomic_transfer(from, to, amount)
    local balance = redis.call('GET', from)
    balance = tonumber(balance) or 0
    
    if balance < amount then
        return {err = "insufficient balance"}
    end
    
    redis.call('DECRBY', from, amount)
    redis.call('INCRBY', to, amount)
    
    return {ok = "transfer successful"}
end

-- 执行
return atomic_transfer(KEYS[1], KEYS[2], tonumber(ARGV[1]))

4.1 分布式锁实现 ​

4.1.1 基础分布式锁 ​

lua
-- 获取锁
-- KEYS[1]: 锁key
-- ARGV[1]: 锁值
-- ARGV[2]: 过期时间（毫秒）
local lock_set = function()
    return redis.call('SET', KEYS[1], ARGV[1], 'NX', 'PX', ARGV[2])
end

-- 释放锁（检查锁值）
local lock_release = function()
    if redis.call('GET', KEYS[1]) == ARGV[1] then
        return redis.call('DEL', KEYS[1])
    end
    return 0
end

-- 续期锁
local lock_renew = function()
    if redis.call('GET', KEYS[1]) == ARGV[1] then
        return redis.call('PEXPIRE', KEYS[1], ARGV[2])
    end
    return 0
end

4.1.2 RedLock算法实现 ​

lua
-- RedLock Lua实现（简化版）
local function redlock_acquire(keys, value, ttl)
    local quorum = math.floor(#keys / 2) + 1
    local acquired = 0
    local start_time = redis.call('TIME')[1] * 1000 + redis.call('TIME')[2] / 1000
    
    for i, key in ipairs(keys) do
        if redis.call('SET', key, value, 'NX', 'PX', ttl) then
            acquired = acquired + 1
        end
    end
    
    local elapsed = (redis.call('TIME')[1] * 1000 + redis.call('TIME')[2] / 1000) - start_time
    local valid_ttl = ttl - elapsed
    
    if acquired >= quorum and valid_ttl > 0 then
        return {valid_ttl, acquired}
    else
        -- 释放所有锁
        for i, key in ipairs(keys) do
            if redis.call('GET', key) == value then
                redis.call('DEL', key)
            end
        end
        return nil
    end
end

4.2 限流器实现 ​

4.2.1 令牌桶算法 ​

lua
-- 令牌桶限流
-- KEYS[1]: 限流key
-- ARGV[1]: 桶容量
-- ARGV[2]: 令牌生成速率（令牌/秒）
-- ARGV[3]: 请求令牌数
local function token_bucket()
    local capacity = tonumber(ARGV[1])
    local rate = tonumber(ARGV[2])
    local requested = tonumber(ARGV[3])
    
    local bucket = redis.call('HMGET', KEYS[1], 'tokens', 'last_refill')
    local tokens = tonumber(bucket[1]) or capacity
    local last_refill = tonumber(bucket[2]) or 0
    
    local now = redis.call('TIME')
    local current_time = now[1] + now[2] / 1000000
    
    -- 计算新增令牌
    if last_refill > 0 then
        local elapsed = current_time - last_refill
        local new_tokens = elapsed * rate
        tokens = math.min(capacity, tokens + new_tokens)
    end
    
    -- 检查是否有足够令牌
    if tokens >= requested then
        tokens = tokens - requested
        redis.call('HMSET', KEYS[1], 'tokens', tokens, 'last_refill', current_time)
        redis.call('EXPIRE', KEYS[1], math.ceil(capacity / rate) * 2)
        return 1  -- 允许
    else
        return 0  -- 拒绝
    end
end

4.2.2 滑动窗口限流 ​

lua
-- 滑动窗口限流
-- KEYS[1]: 限流key
-- ARGV[1]: 窗口大小（秒）
-- ARGV[2]: 最大请求数
local function sliding_window()
    local window = tonumber(ARGV[1])
    local limit = tonumber(ARGV[2])
    
    local now = redis.call('TIME')
    local current_ms = now[1] * 1000 + now[2] / 1000
    local window_start = current_ms - window * 1000
    
    -- 移除过期记录
    redis.call('ZREMRANGEBYSCORE', KEYS[1], 0, window_start)
    
    -- 获取当前窗口内请求数
    local current = redis.call('ZCARD', KEYS[1])
    
    if current < limit then
        -- 添加当前请求
        redis.call('ZADD', KEYS[1], current_ms, current_ms)
        redis.call('EXPIRE', KEYS[1], window)
        return 1
    else
        return 0
    end
end

local function batch_hash_update(hash_key, updates)
    local results = {}

-- 原子执行集合操作并存储结果
local function set_operations(operation, destination, keys)
    local result
    
    if operation == 'INTER' then
        result = redis.call('SINTER', unpack(keys))
    elseif operation == 'UNION' then
        result = redis.call('SUNION', unpack(keys))
    elseif operation == 'DIFF' then
        result = redis.call('SDIFF', unpack(keys))
    else
        return {err = "Invalid operation"}
    end
    
    -- 存储结果
    if #result > 0 then
        redis.call('DEL', destination)
        redis.call('SADD', destination, unpack(result))
    end
    
    return {count = #result, members = result}
end

// user_tracking.c - 用户行为追踪模块
#include "redismodule.h"

#define MAX_EVENTS 1000

typedef struct {
    char *user_id;
    char *event_type;
    char *item_id;
    time_t timestamp;
    double score;
} UserEvent;

typedef struct {
    UserEvent *events;
    size_t count;
    size_t capacity;
} UserProfile;

// 添加用户事件
int AddUserEvent_RedisCommand(RedisModuleCtx *ctx, RedisModuleString **argv, int argc) {
    if (argc != 5) {
        return RedisModule_WrongArity(ctx);
    }
    
    // 获取参数
    const char *user_id = RedisModule_StringPtrLen(argv[1], NULL);
    const char *event_type = RedisModule_StringPtrLen(argv[2], NULL);
    const char *item_id = RedisModule_StringPtrLen(argv[3], NULL);
    double score;
    RedisModule_StringToDouble(argv[4], &score);
    
    // 获取或创建用户画像
    RedisModuleKey *key = RedisModule_OpenKey(ctx, argv[1], REDISMODULE_READ | REDISMODULE_WRITE);
    UserProfile *profile;
    
    if (RedisModule_KeyType(key) == REDISMODULE_KEYTYPE_EMPTY) {
        // 创建新画像
        profile = RedisModule_Alloc(sizeof(UserProfile));
        profile->capacity = 100;
        profile->events = RedisModule_Alloc(profile->capacity * sizeof(UserEvent));
        profile->count = 0;
        
        RedisModule_ModuleTypeSetValue(key, UserProfileType, profile);
    } else {
        profile = RedisModule_ModuleTypeGetValue(key);
    }
    
    // 添加事件
    if (profile->count >= profile->capacity) {
        // 扩容
        profile->capacity *= 2;
        profile->events = RedisModule_Realloc(profile->events, 
                                              profile->capacity * sizeof(UserEvent));
    }
    
    UserEvent *event = &profile->events[profile->count++];
    event->user_id = RedisModule_Strdup(user_id);
    event->event_type = RedisModule_Strdup(event_type);
    event->item_id = RedisModule_Strdup(item_id);
    event->timestamp = time(NULL);
    event->score = score;
    
    // 更新推荐分数
    UpdateRecommendationScore(ctx, user_id, item_id, score);
    
    RedisModule_ReplyWithSimpleString(ctx, "OK");
    RedisModule_CloseKey(key);
    return REDISMODULE_OK;
}

十二、总结 ​

12.1 技术选型建议 ​

Lua脚本适用场景： ​

  1. 简单业务逻辑：原子操作、数据校验
  2. 快速原型：无需编译部署
  3. 中小规模应用：性能要求不极致
  4. 维护简单：团队熟悉Lua

Redis模块适用场景： ​

  1. 高性能要求：需要极致性能
  2. 复杂数据结构：自定义数据类型
  3. 系统扩展：需要新命令和功能
  4. 企业级应用：有专业C/C++开发团队

12.2 最佳实践总结 ​

  1. 安全性优先：始终验证输入，限制资源使用
  2. 性能监控：持续监控脚本和模块性能
  3. 版本管理：做好脚本和模块的版本控制
  4. 文档齐全：为自定义函数和模块编写文档
  5. 测试完备：单元测试和集成测试必不可少

12.3 学习路径建议 ​

  1. 初学者：从Lua脚本开始，掌握基础命令
  2. 中级开发者：学习复杂Lua模式，了解模块基础
  3. 高级开发者：深入模块开发，理解Redis内部机制
  4. 架构师：设计基于Redis扩展的分布式系统

通过合理使用Lua脚本和Redis模块，可以在保持Redis高性能的同时，实现复杂的业务逻辑和定制功能，为各种应用场景提供强大的数据服务能力。
