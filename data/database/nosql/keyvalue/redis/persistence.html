一、Redis持久化概述 ​

1.1 为什么需要持久化？ ​

Redis是内存数据库，数据存储在内存中。持久化解决的核心问题：

  • 数据持久保存：防止服务器重启或崩溃导致数据丢失
  • 灾难恢复：从持久化文件中恢复数据
  • 数据备份：用于数据迁移、复制等场景
  • 数据一致性：在分布式环境中保证数据可靠性

1.2 Redis持久化架构 ​

┌─────────────────────────────────────┐
│          Redis Server                │
├─────────────────────────────────────┤
│       数据操作层 (内存数据库)         │
├─────────────────────────────────────┤
│  │              │              │    │
│  ▼              ▼              ▼    │
│ RDB持久化      AOF持久化     混合持久化  │
├─────────────────────────────────────┤
│         操作系统文件系统              │
└─────────────────────────────────────┘

二、RDB持久化机制 ​

2.1 RDB概述 ​

RDB（Redis DataBase）是Redis的快照式持久化，在指定时间间隔将内存中的数据集快照写入磁盘。

2.2 RDB文件结构 ​

┌─────────────────┬─────────────────┬──────────────┬──────────┐
│  REDIS  (5字节) │  RDB版本 (4字节) │  AUX字段      │  DB数据   │ ...
│  "REDIS"        │  "0009"         │  key-value对 │  数据库内容 │
└─────────────────┴─────────────────┴──────────────┴──────────┘

数据库内容结构：
┌─────────────┬──────────────────┬──────────────────┬────────────┐
│ SELECTDB    │ 数据库编号        │ 键值对数量       │ 过期键值对  │
│ 0xFE        │ (变长编码)        │ (变长编码)       │  特殊操作   │
└─────────────┴──────────────────┴──────────────────┴────────────┘

2.3 RDB触发机制 ​

2.3.1 自动触发 ​

配置文件设置（redis.conf）：

yaml
# 满足以下任一条件即触发BGSAVE
save 900 1      # 900秒内至少有1个key被修改
save 300 10     # 300秒内至少有10个key被修改  
save 60 10000   # 60秒内至少有10000个key被修改

# 其他相关配置
dbfilename dump.rdb        # RDB文件名
dir ./                     # 存储目录
rdbcompression yes         # 是否压缩（LZF算法）
rdbchecksum yes           # 是否校验和
stop-writes-on-bgsave-error yes  # BGSAVE出错时停止写入

2.3.2 手动触发 ​

bash
# 1. SAVE命令（同步，会阻塞）
redis-cli> SAVE  # 阻塞直到RDB创建完成

# 2. BGSAVE命令（异步，非阻塞）
redis-cli> BGSAVE  # 后台执行，立即返回"Background saving started"

# 3. 通过配置文件自动触发
# 4. SHUTDOWN命令（正常关闭时自动执行SAVE）
# 5. 主从复制时，从节点连接会触发主节点执行BGSAVE

2.4 RDB执行流程 ​

2.4.1 BGSAVE实现原理 ​

主进程                              子进程
  │                                   │
  │ 1. 接收BGSAVE命令                  │
  │ 2. 检查是否有其他持久化进程运行    │
  │ 3. 调用fork()创建子进程            │──────┐
  │ 4. 继续处理客户端请求              │      │
  │    │                              │      │
  │    │                              │ 5. 子进程开始工作
  │    │                              │ 6. 遍历所有数据库
  │    │                              │ 7. 将数据写入临时RDB文件
  │    │                              │ 8. 原子替换旧RDB文件
  │    │                              │ 9. 向父进程发送信号
  │    │◄─────────────────────────────│
  │ 10. 处理子进程完成信号            │
  │ 11. 更新持久化状态信息            │
  │                                   │

2.4.2 核心源码分析 ​

c
// src/rdb.c
void saveCommand(client *c) {
    // SAVE命令同步执行
    if (server.rdb_child_pid != -1) {
        addReplyError(c,"Background save already in progress");
        return;
    }
    rdbSaveInfo rsi, *rsiptr;
    rsiptr = rdbPopulateSaveInfo(&rsi);
    if (rdbSave(server.rdb_filename,rsiptr) == C_OK) {
        addReply(c,shared.ok);
    } else {
        addReply(c,shared.err);
    }
}

void bgsaveCommand(client *c) {
    // BGSAVE命令异步执行
    if (server.rdb_child_pid != -1) {
        addReplyError(c,"Background save already in progress");
    } else if (server.aof_child_pid != -1) {
        addReplyError(c,"Can't BGSAVE while AOF log rewriting is in progress");
    } else if (rdbSaveBackground(server.rdb_filename,NULL) == C_OK) {
        addReplyStatus(c,"Background saving started");
    } else {
        addReply(c,shared.err);
    }
}

int rdbSaveBackground(char *filename, rdbSaveInfo *rsi) {
    pid_t childpid;
    
    if (server.aof_child_pid != -1 || server.rdb_child_pid != -1) return C_ERR;
    
    server.dirty_before_bgsave = server.dirty;
    server.lastbgsave_try = time(NULL);
    
    if ((childpid = redisFork(CHILD_TYPE_RDB)) == 0) {
        // 子进程
        redisSetProcTitle("redis-rdb-bgsave");
        redisSetCpuAffinity(server.bgsave_cpulist);
        
        // 执行RDB保存
        if (rdbSave(filename,rsi) == C_OK) {
            size_t private_dirty = zmalloc_get_private_dirty(-1);
            if (private_dirty) {
                serverLog(LL_NOTICE,
                    "RDB: %zu MB of memory used by copy-on-write",
                    private_dirty/(1024*1024));
            }
            exitFromChild(0);
        } else {
            exitFromChild(1);
        }
    } else {
        // 父进程
        if (childpid == -1) {
            server.lastbgsave_status = C_ERR;
            serverLog(LL_WARNING,"Can't save in background: fork: %s",
                strerror(errno));
            return C_ERR;
        }
        serverLog(LL_NOTICE,"Background saving started by pid %ld",(long) childpid);
        server.rdb_child_pid = childpid;
        server.rdb_child_type = RDB_CHILD_TYPE_DISK;
        updateDictResizePolicy();
        return C_OK;
    }
    return C_OK; /* unreached */
}

2.5 写时复制（Copy-On-Write）技术 ​

2.5.1 COW原理 ​

进程A内存空间                    进程B内存空间
┌─────────────────┐              ┌─────────────────┐
│ 页表            │ fork()       │ 复制页表         │
│ 指向物理页      │─────────────▶│ 指向相同物理页    │
└─────────────────┘              └─────────────────┘
         │                               │
         │ 共享相同的物理内存页           │
         ▼                               ▼
┌─────────────────┐              ┌─────────────────┐
│ 物理内存页       │◀─────────────│ 相同的物理内存页  │
│ (实际数据)       │               │                 │
└─────────────────┘              └─────────────────┘

当进程A修改数据时：
┌─────────────────┐              ┌─────────────────┐
│ 页表            │              │ 页表            │
│ 指向新的物理页   │              │ 仍指向旧物理页   │
└─────────────────┘              └─────────────────┘
         │                               │
         ▼                               ▼
┌─────────────────┐              ┌─────────────────┐
│ 新物理内存页     │               │ 旧物理内存页     │
│ (修改后数据)     │               │ (原始数据)       │
└─────────────────┘              └─────────────────┘

2.5.2 COW的内存影响 ​

  • 内存增长：子进程保存期间，如果父进程修改数据，会复制内存页
  • 内存峰值：最坏情况下，内存占用可能翻倍
  • 优化建议：
    • 避免在BGSAVE期间大量写入
    • 使用vm.overcommit_memory = 1（Linux）
    • 监控used_memory和used_memory_peak

2.7 RDB优缺点分析 ​

优点： ​

  1. 紧凑的二进制格式：文件小，适合备份和传输
  2. 快速恢复：直接加载到内存，恢复速度快
  3. 最大化性能：父进程不需要磁盘I/O
  4. 适合灾难恢复：完整的数据库快照

缺点： ​

  1. 数据丢失风险：两次快照之间的数据可能丢失
  2. fork性能问题：大数据集时fork可能阻塞
  3. 文件一致性：RDB文件可能不完整（崩溃时）

三、AOF持久化机制 ​

3.1 AOF概述 ​

AOF（Append Only File）通过记录所有写操作命令来持久化数据，类似于数据库的写前日志（WAL）。

3.2 AOF文件结构 ​

*3              # 3个元素
$3              # 第一个元素长度3
SET             # 命令
$5              # 键长度
mykey           # 键
$7              # 值长度
myvalue         # 值

实际AOF文件内容：
*3\r\n$3\r\nSET\r\n$5\r\nmykey\r\n$7\r\nmyvalue\r\n

3.3 AOF工作流程 ​

┌─────────────────────────────────────────────────────────┐
│                    Redis Server                          │
├─────────────────────────────────────────────────────────┤
│  执行命令 → 命令传播到AOF → 写入AOF缓冲区 → 同步到磁盘     │
└─────────────────────────────────────────────────────────┘
         │              │              │           │
         │              │              │           │
         ▼              ▼              ▼           ▼
┌─────────────────────────────────────────────────────────┐
│  处理结果   AOF命令      aof_buf缓冲区   AOF文件          │
└─────────────────────────────────────────────────────────┘

3.4 AOF持久化策略 ​

3.4.1 三种刷盘策略 ​

yaml
# redis.conf配置
appendonly yes                    # 开启AOF
appendfilename "appendonly.aof"   # AOF文件名
appendfsync everysec              # 刷盘策略

# 策略说明：
# 1. appendfsync always     每个命令都同步，最安全，性能最差
# 2. appendfsync everysec   每秒同步一次，平衡方案（默认）
# 3. appendfsync no         由操作系统决定，性能最好，最不安全

3.4.2 策略实现源码 ​

c
// src/aof.c
void flushAppendOnlyFile(int force) {
    ssize_t nwritten;
    int sync_in_progress = 0;
    
    if (sdslen(server.aof_buf) == 0) return;
    
    if (server.aof_fsync == AOF_FSYNC_EVERYSEC)
        sync_in_progress = bioPendingJobsOfType(BIO_AOF_FSYNC) != 0;
    
    // 写入操作系统缓冲区
    nwritten = aofWrite(server.aof_fd,server.aof_buf,sdslen(server.aof_buf));
    
    // 根据策略同步到磁盘
    if (server.aof_fsync == AOF_FSYNC_ALWAYS) {
        // always策略：立即同步
        redis_fsync(server.aof_fd);
        server.aof_last_fsync = server.unixtime;
    } else if (server.aof_fsync == AOF_FSYNC_EVERYSEC) {
        // everysec策略：异步同步
        if (!sync_in_progress) aof_background_fsync(server.aof_fd);
        server.aof_last_fsync = server.unixtime;
    }
    // no策略：不主动同步，由操作系统决定
}

3.5 AOF重写（Rewrite）机制 ​

3.5.1 为什么需要AOF重写？ ​

  • AOF文件持续增长，可能占用大量磁盘空间
  • 恢复时间变长
  • 包含冗余命令（如多次SET同一个key）

3.5.2 AOF重写触发条件 ​

yaml

auto-aof-rewrite-min-size 64mb    # AOF文件至少64MB才重写

# 手动触发
redis-cli> BGREWRITEAOF

3.5.3 AOF重写流程 ​

主进程                              子进程
  │                                   │
  │ 1. 接收BGREWRITEAOF命令            │
  │ 2. fork()创建子进程                │──────┐
  │ 3. 继续处理客户端请求              │      │
  │    │                              │ 4. 子进程开始重写
  │    │ 5. 写入重写缓冲区             │ 5. 读取内存数据快照
  │    │ (新命令)                      │ 6. 生成新AOF临时文件
  │    │◄─────────────────────────────│ 7. 完成，通知父进程
  │ 8. 将重写缓冲区的命令追加到新文件   │      │
  │ 9. 原子替换旧AOF文件               │      │
  │ 10. 继续将新命令写入新AOF文件       │      │
  │                                   │      │

3.5.4 AOF重写源码解析 ​

c
// src/aof.c
int rewriteAppendOnlyFile(char *filename) {
    rio aof;
    FILE *fp;
    char tmpfile[256];
    long long now = mstime();
    
    // 创建临时文件
    snprintf(tmpfile,256,"temp-rewriteaof-%d.aof", (int) getpid());
    fp = fopen(tmpfile,"w");
    
    // 初始化rio结构
    rioInitWithFile(&aof,fp);
    
    // 设置AOF重写处理函数
    if (rioWrite(&aof,"REDIS",5) == 0) goto werr;
    
    // 遍历所有数据库
    for (j = 0; j < server.dbnum; j++) {
        // 写入SELECT DB命令
        // 遍历数据库中的所有key
        while((de = dictNext(di)) != NULL) {
            sds keystr = dictGetKey(de);
            robj key, *o = dictGetVal(de);
            long long expiretime = getExpire(db,&key);
            
            // 根据数据类型生成最小命令集
            if (o->type == OBJ_STRING) {
                // 生成SET命令
                emitSETKey(&aof,dbid,keystr,o,expiretime);
            } else if (o->type == OBJ_LIST) {
                // 生成RPUSH命令
                rewriteListObject(&aof,keystr,o,expiretime);
            }
            // ... 其他数据类型
        }
    }
    
    fclose(fp);
    return C_OK;
}

3.8 AOF优缺点分析 ​

优点： ​

  1. 数据安全：最多丢失1秒数据（everysec策略）
  2. 可读性强：文本格式，便于调试和修复
  3. 容错性好：AOF文件损坏可通过工具修复
  4. 灵活的策略：可根据需求调整同步频率

缺点： ​

  1. 文件体积大：相同数据集比RDB文件大
  2. 恢复速度慢：需要重新执行所有命令
  3. 写入性能影响：AOF同步对性能有一定影响
  4. 重写期间内存占用：fork子进程可能导致内存翻倍

四、混合持久化（RDB + AOF） ​

4.1 混合持久化概述 ​

Redis 4.0+引入，结合RDB和AOF的优点：

  • 快速加载：使用RDB格式作为数据基础
  • 低丢失风险：AOF记录增量变化

4.2 混合持久化文件结构 ​

┌─────────────────────────────────────────────────────┐
│                    AOF文件                            │
├─────────────────────────────────────────────────────┤
│  RDB格式数据     │          AOF格式增量命令             │
│  (二进制)        │          (文本协议)                 │
└─────────────────────────────────────────────────────┘
↑                   ↑
RDB部分              AOF部分

4.3 混合持久化配置 ​

yaml
# 开启AOF
appendonly yes

# 开启混合持久化（Redis 4.0+）
aof-use-rdb-preamble yes

# AOF重写时自动使用混合格式
# 重写后的AOF文件包含：
# 1. RDB格式的全量数据（快速加载）
# 2. 后续的AOF格式增量命令

4.4 混合持久化加载流程 ​

c
// src/aof.c - 加载AOF文件
int loadAppendOnlyFile(char *filename) {
    // 检查文件头部
    char sig[5];
    if (fread(sig,1,5,fp) != 5 || memcmp(sig,"REDIS",5) != 0) {
        // 旧版AOF格式，按文本协议解析
    } else {
        // 检查版本
        if (fread(&version,1,4,fp) != 4) return -1;
        
        // 检查是否是混合持久化格式
        if (version[0] == 'R' && version[1] == 'D' && 
            version[2] == 'B' && version[3] == '\0') {
            // RDB格式部分
            rio rdb;
            rioInitWithFile(&rdb,fp);
            if (rdbLoadRio(&rdb,RDBFLAGS_AOF_PREAMBLE,NULL) != C_OK) {
                serverLog(LL_WARNING,"Error reading the RDB preamble of the AOF file");
                return -1;
            }
        } else {
            // 旧版AOF格式
            fseek(fp,0,SEEK_SET);
        }
        
        // 继续加载AOF格式部分
        while(1) {
            // 解析AOF命令并执行
        }
    }
}

4.5 混合持久化优缺点 ​

优点： ​

  1. 快速恢复：RDB部分提供快速加载
  2. 数据安全：AOF部分保证数据完整性
  3. 兼容性好：向后兼容旧版AOF格式

缺点： ​

  1. 文件可读性差：RDB部分是二进制，不可直接查看
  2. 实现复杂度高：需要处理两种格式的切换

  2. 避免大数据集频繁保存：减少fork开销

  2. 优化AOF重写触发条件：

九、总结 ​

Redis持久化机制提供了灵活可靠的数据保护方案：

关键要点： ​

  1. RDB适合备份和快速恢复，但可能丢失数据
  2. AOF提供更好的数据安全性，但文件较大
  3. 混合持久化结合两者优点，是推荐的生产环境配置
  4. 持久化配置需要根据业务需求调整，平衡性能和数据安全

最佳实践原则： ​

  1. 理解业务需求：根据数据重要性选择策略
  2. 监控持久化状态：定期检查持久化健康状况
  3. 制定备份策略：定期备份并测试恢复流程
  4. 容量规划：预留足够的磁盘空间和内存
  5. 性能测试：在生产前测试持久化对性能的影响

故障处理黄金法则： ​

  1. 总是备份持久化文件
  2. 优先尝试AOF修复
  3. 监控fork时间和持久化状态
  4. 准备好手动恢复流程

通过合理配置和监控，Redis持久化可以为企业级应用提供可靠的数据保护，在性能和安全性之间取得最佳平衡。
