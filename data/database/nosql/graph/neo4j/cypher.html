Neo4j Cypher查询语言学习笔记 ​

一、Cypher语言概述 ​

1.1 Cypher设计哲学 ​

1.1.1 声明式查询语言 ​

┌─────────────────────────────────────────────┐
│             Cypher的核心特点                 │
├─────────────────────────────────────────────┤
│ • 描述"想要什么"而不是"如何获取"            │
│ • 基于模式匹配(Pattern Matching)            │
│ • 人类可读的ASCII艺术语法                  │
│ • 专为图数据操作设计                       │
└─────────────────────────────────────────────┘

与SQL对比示例：

cypher
-- Cypher: 直观的模式描述
MATCH (p:Person)-[:FRIEND]->(friend)
WHERE p.name = "Alice"
RETURN friend.name

-- SQL: 需要连接表
SELECT f.name 
FROM person p
JOIN friendship fs ON p.id = fs.person_id
JOIN person f ON fs.friend_id = f.id
WHERE p.name = 'Alice'

1.1.2 ASCII艺术语法 ​

cypher
// 节点表示：括号 ()
(p:Person {name: "Alice"})

// 关系表示：方括号 []
-[:KNOWS {since: 2010}]->

// 路径表示：组合模式
path = (a)-[:KNOWS*1..3]->(b)

1.2 Cypher执行模型 ​

1.2.1 查询处理流程 ​

输入Cypher查询
    ↓
解析与语法检查
    ↓
生成执行计划
    ↓
优化器优化
    ↓
运行时执行
    ↓
返回结果

1.2.2 执行计划分析 ​

cypher
-- 查看执行计划(不执行)
EXPLAIN MATCH (p:Person) RETURN p

-- 执行并分析性能
PROFILE MATCH (p:Person)-[:KNOWS]->(f)
RETURN p.name, count(f) as friendCount
ORDER BY friendCount DESC

二、Cypher基础语法 ​

2.1 数据定义语言(DDL) ​

2.1.1 节点操作 ​

cypher
-- 创建单个节点
CREATE (:Person {name: "Alice", age: 30})

-- 创建多个节点
CREATE 
  (:Person {name: "Bob", age: 25}),
  (:Person {name: "Charlie", age: 35}),
  (:Company {name: "TechCorp"})

-- 创建带多个标签的节点
CREATE (p:Person:Employee:Manager {
  employeeId: "E001",
  name: "David",
  department: "Engineering",
  hireDate: date("2020-06-15")
})

-- 从查询创建节点
MATCH (c:Company {name: "TechCorp"})
CREATE (p:Person {name: "Eve"})-[:WORKS_AT]->(c)
RETURN p, c

2.1.2 关系操作 ​

cypher
-- 创建关系(节点已存在)
MATCH (a:Person {name: "Alice"})
MATCH (b:Person {name: "Bob"})
CREATE (a)-[r:KNOWS {
  since: 2015,
  metAt: "University",
  closeness: 0.8
}]->(b)
RETURN r

-- 一次性创建节点和关系
CREATE 
  (a:Person {name: "Frank"})-[:FRIEND_OF {
    since: date("2021-03-10"),
    status: "close"
  }]->(b:Person {name: "Grace"}),
  (a)-[:COLLEAGUE]->(c:Person {name: "Henry"})

-- 创建双向关系
CREATE 
  (a)-[:FRIEND]->(b),
  (b)-[:FRIEND]->(a)

-- 创建自引用关系
MATCH (p:Person {name: "Alice"})
CREATE (p)-[:MANAGES]->(p)

2.2 数据查询语言(DQL) ​

2.2.1 基础查询 ​

cypher
-- 返回所有节点
MATCH (n) RETURN n

-- 返回特定类型节点
MATCH (p:Person) RETURN p

-- 返回属性
MATCH (p:Person) 
RETURN p.name, p.age, p.email

-- 使用别名
MATCH (p:Person) 
RETURN p.name AS fullName, p.age AS yearsOld

-- 去重
MATCH (p:Person)-[:WORKS_IN]->(d:Department)
RETURN DISTINCT d.name

2.2.2 条件过滤 ​

cypher
-- WHERE子句
MATCH (p:Person)
WHERE p.age > 25 AND p.age < 40
RETURN p.name, p.age

-- 属性存在性检查
MATCH (p:Person)
WHERE EXISTS(p.email) AND p.email IS NOT NULL
RETURN p

-- 字符串匹配
MATCH (p:Person)
WHERE p.name STARTS WITH 'A'
   OR p.name ENDS WITH 'son'
   OR p.name CONTAINS 'li'
RETURN p.name

-- 正则表达式
MATCH (p:Person)
WHERE p.name =~ 'J.*n'  -- 以J开头，n结尾
   OR p.email =~ '.*@gmail\\.com$'
RETURN p

-- 集合操作
MATCH (p:Person)
WHERE p.name IN ["Alice", "Bob", "Charlie"]
   AND p.age IN range(20, 30)  -- 20到30之间
RETURN p

-- NULL值处理
MATCH (p:Person)
WHERE p.middleName IS NULL
RETURN p.name

2.2.3 排序和分页 ​

cypher
-- 排序
MATCH (p:Person)
RETURN p.name, p.age
ORDER BY p.age DESC, p.name ASC

-- 分页
MATCH (p:Person)
RETURN p
SKIP 20      -- 跳过前20条
LIMIT 10     -- 返回10条

-- 组合使用
MATCH (p:Person)
WHERE p.department = "Engineering"
RETURN p.name, p.salary
ORDER BY p.salary DESC
SKIP 0 LIMIT 50

2.3 数据操作语言(DML) ​

2.3.1 更新操作 ​

cypher
-- 设置属性
MATCH (p:Person {name: "Alice"})
SET p.age = 31,
    p.lastUpdated = timestamp(),
    p.updatedBy = "system"

-- 设置多个属性(MAP)
MATCH (p:Person {name: "Bob"})
SET p += {
  age: 26,
  status: "active",
  metadata: {source: "HR", version: 2}
}

-- 添加标签
MATCH (p:Person {name: "Charlie"})
SET p:Manager:Senior

-- 移除标签
MATCH (p:Person:Manager {name: "David"})
REMOVE p:Manager
SET p:FormerManager

-- 移除属性
MATCH (p:Person {name: "Eve"})
REMOVE p.temporaryNote

-- 复制属性
MATCH (p1:Person {name: "Alice"})
MATCH (p2:Person {name: "Bob"})
SET p2.department = p1.department

2.3.2 删除操作 ​

cypher
-- 删除节点(必须先删除关系)
MATCH (p:Person {name: "Obsolete"})
DETACH DELETE p

-- 删除关系
MATCH (a:Person {name: "Alice"})-[r:OLD_RELATION]-(b:Person)
DELETE r

-- 批量删除
MATCH (p:Person)
WHERE p.lastLogin < datetime().subtract(Duration.ofDays(365))
DETACH DELETE p

2.3.3 合并操作(MERGE) ​

cypher
-- 节点合并(存在则更新，不存在则创建)
MERGE (p:Person {email: "alice@example.com"})
ON CREATE SET 
  p.created = timestamp(),
  p.name = "Alice",
  p.status = "new"
ON MATCH SET 
  p.lastSeen = timestamp(),
  p.loginCount = coalesce(p.loginCount, 0) + 1
RETURN p

-- 关系合并
MATCH (a:Person {name: "Alice"})
MATCH (b:Person {name: "Bob"})
MERGE (a)-[r:KNOWS]-(b)
ON CREATE SET r.since = date()
ON MATCH SET r.lastMet = date()
RETURN r

-- 完整模式合并
MERGE (a:Person {name: "Alice"})-[:FRIEND]->(b:Person {name: "Bob"})
ON CREATE SET a.created = timestamp(), b.created = timestamp()

三、高级查询模式 ​

3.1 路径查询 ​

3.1.1 基本路径匹配 ​

cypher
-- 固定长度路径
MATCH (a:Person {name: "Alice"})-[:KNOWS]-(b:Person {name: "Bob"})
RETURN a, b

-- 可变长度路径
MATCH (a:Person {name: "Alice"})-[:KNOWS*1..3]-(friend)
RETURN DISTINCT friend.name

-- 精确长度路径
MATCH (a:Person {name: "Alice"})-[:KNOWS*2]-(friendOfFriend)
RETURN friendOfFriend.name

-- 无限长度路径(谨慎使用)
MATCH (a:Person {name: "Alice"})-[:KNOWS*]-(connected)
RETURN connected.name

3.1.2 路径变量和函数 ​

cypher
-- 捕获路径
MATCH path = (a:Person)-[:KNOWS*1..3]-(b:Person)
RETURN path, length(path) as depth

-- 路径节点和关系访问
MATCH path = (a:Person)-[:KNOWS*1..3]-(b)
RETURN 
  nodes(path) as pathNodes,
  relationships(path) as pathRels,
  [node IN nodes(path) | node.name] as names

-- 最短路径
MATCH (a:Person {name: "Alice"}), (b:Person {name: "Zach"}),
path = shortestPath((a)-[:KNOWS*]-(b))
RETURN path, length(path)

-- 带约束的最短路径
MATCH (a:Person {name: "Alice"}), (b:Person {name: "Bob"}),
path = shortestPath((a)-[:KNOWS|FRIEND*]-(b))
WHERE all(r IN relationships(path) WHERE r.since > date("2020-01-01"))
RETURN path

-- 所有最短路径
MATCH (a:Person), (b:Person),
paths = allShortestPaths((a)-[:KNOWS*]-(b))
WHERE a.name = "Alice" AND b.name = "Bob"
RETURN paths

3.2 聚合函数 ​

3.2.1 基本聚合 ​

cypher
-- 计数
MATCH (p:Person)
RETURN count(p) as totalPeople

-- 分组计数
MATCH (p:Person)-[:WORKS_IN]->(d:Department)
RETURN d.name, count(p) as employeeCount

-- 统计函数
MATCH (p:Person)
RETURN 
  avg(p.age) as averageAge,
  min(p.age) as youngest,
  max(p.age) as oldest,
  sum(p.age) as totalAge,
  stDev(p.age) as ageDeviation

-- 收集为列表
MATCH (p:Person)-[:BELONGS_TO]->(t:Team)
RETURN t.name, collect(p.name) as members

-- 收集去重
MATCH (p:Person)
RETURN collect(DISTINCT p.department) as uniqueDepartments

3.2.2 高级聚合 ​

cypher
-- 分组和过滤(HAVING)
MATCH (p:Person)-[:WORKS_IN]->(d:Department)
WITH d, count(p) as count
WHERE count > 5
RETURN d.name, count
ORDER BY count DESC

-- 多级聚合
MATCH (c:Company)<-[:WORKS_AT]-(p:Person)
WITH c, p.department as dept, count(p) as deptCount
RETURN c.name, collect({department: dept, count: deptCount}) as departments

-- 百分比计算
MATCH (p:Person)
WITH count(p) as total
MATCH (p:Person)
WHERE p.gender = "Female"
RETURN count(p) as femaleCount, 
       round(100.0 * count(p) / total, 2) as percentage

3.3 子查询和WITH子句 ​

3.3.1 WITH子句用法 ​

cypher
-- 管道式数据处理
MATCH (p:Person)
WITH p
WHERE p.age > 18
WITH p
ORDER BY p.name
RETURN collect(p.name) as adultNames

-- 聚合后继续查询
MATCH (p:Person)-[:WORKS_IN]->(d:Department)
WITH d, count(p) as size
WHERE size > 10
MATCH (d)-[:LOCATED_IN]->(city:City)
RETURN d.name, city.name, size

-- 限制中间结果
MATCH (p:Person)
WITH p
ORDER BY p.joinDate DESC
LIMIT 100
MATCH (p)-[:HAS_SKILL]->(s:Skill)
RETURN p.name, collect(s.name) as recentHireSkills

3.3.2 子查询模式 ​

cypher
-- 使用CALL进行子查询(Neo4j 4.0+)
MATCH (p:Person)
CALL {
  WITH p
  MATCH (p)-[:FRIEND]->(f:Person)
  RETURN count(f) as friendCount
}
RETURN p.name, friendCount

-- 相关子查询
MATCH (company:Company)
CALL {
  WITH company
  MATCH (company)<-[:WORKS_AT]-(p:Person)
  RETURN avg(p.salary) as avgSalary
}
RETURN company.name, avgSalary

-- 多个子查询
MATCH (p:Person {name: "Alice"})
CALL {
  WITH p
  MATCH (p)-[:FRIEND]->(f:Person)
  RETURN f
}
CALL {
  WITH p
  MATCH (p)-[:LIKES]->(post:Post)
  RETURN post
}
RETURN p, collect(f) as friends, collect(post) as likedPosts

4.2.2 列表推导式(List Comprehension) ​

cypher
-- 基本推导式
MATCH (p:Person)
RETURN [person IN collect(p) | person.name] as names

-- 带条件的推导式
MATCH (p:Person)
RETURN [person IN collect(p) WHERE person.age > 30 | person.name] as olderNames

-- 转换元素
WITH [1, 2, 3, 4, 5] as numbers
RETURN [x IN numbers | x * x] as squares  -- [1, 4, 9, 16, 25]

-- 复杂推导式
MATCH (p:Person)-[:HAS_SKILL]->(s:Skill)
WITH p, collect(s) as skills
RETURN p.name, 
       [skill IN skills WHERE skill.level > 3 | skill.name] as advancedSkills

4.2.3 列表函数 ​

cypher
-- 列表操作函数
WITH [1, 2, 3, 4, 5] as list
RETURN 
  head(list) as first,        -- 1
  last(list) as last,         -- 5
  tail(list) as rest,         -- [2, 3, 4, 5]
  reverse(list) as reversed,  -- [5, 4, 3, 2, 1]
  [x IN list | x * 2] as doubled

-- 列表聚合
RETURN 
  reduce(total = 0, x IN [1, 2, 3, 4, 5] | total + x) as sum,  -- 15
  reduce(names = "", p IN [{name:"A"}, {name:"B"}] | names + p.name) as combined

-- 列表展开(UNWIND)
UNWIND [1, 2, 3] AS number
RETURN number * 2 as doubled

-- UNWIND应用：展开列表查询
MATCH (p:Person)
WHERE p.name IN ["Alice", "Bob", "Charlie"]
UNWIND ["Math", "Science", "Art"] as subject
CREATE (p)-[:STUDIED {grade: "A"}]->(:Subject {name: subject})

五、模式匹配高级技巧 ​

5.1 多重模式匹配 ​

cypher
-- 多个独立模式
MATCH (a:Person {name: "Alice"}), (b:Person {name: "Bob"})
RETURN a, b

-- 模式组合
MATCH (a:Person {name: "Alice"})
MATCH (a)-[:FRIEND]->(friend)
MATCH (friend)-[:LIKES]->(post:Post)
RETURN friend.name, collect(post.title) as likedPosts

-- 可选匹配(LEFT JOIN)
MATCH (p:Person)
OPTIONAL MATCH (p)-[:MANAGES]->(team:Team)
RETURN p.name, team.name  -- team.name可能为NULL

-- 模式否定
MATCH (p:Person)
WHERE NOT (p)-[:FRIEND]->(:Person {name: "Bob"})
RETURN p.name

-- 存在性检查
MATCH (p:Person)
WHERE (p)-[:WORKS_AT]->(:Company {name: "TechCorp"})
RETURN p.name

5.2 图遍历控制 ​

cypher
-- 遍历时收集
MATCH path = (a:Person)-[:KNOWS*1..3]->(b)
WITH a, b, 
     [node IN nodes(path) | node.name] as pathNames,
     [rel IN relationships(path) | type(rel)] as relTypes
RETURN a.name, b.name, pathNames, relTypes

-- 避免循环路径
MATCH path = (a:Person)-[:KNOWS*1..5]->(b)
WHERE a <> b AND all(node IN nodes(path) WHERE size([n IN nodes(path) WHERE n = node]) = 1)
RETURN path

-- 特定节点类型约束
MATCH (a:Person)-[:CONNECTED*1..3]->(b)
WHERE all(node IN nodes(path) WHERE node:Person OR node:Company)
RETURN path

六、性能优化和最佳实践 ​

6.1 查询优化技巧 ​

6.1.1 索引和约束利用 ​

cypher
-- 确保使用索引
-- 创建索引
CREATE INDEX person_name_index FOR (p:Person) ON (p.name)
CREATE INDEX person_email_index FOR (p:Person) ON (p.email)

-- 索引自动使用
MATCH (p:Person {name: "Alice"})  -- 使用索引
RETURN p

MATCH (p:Person) WHERE p.name = "Alice"  -- 同样使用索引
RETURN p

-- 复合查询优化
MATCH (p:Person)
WHERE p.name = "Alice" AND p.age > 25  -- 先使用索引过滤name，再过滤age
RETURN p

6.1.2 查询重写优化 ​

cypher
-- 优化前：多次扫描
MATCH (p:Person)
WHERE p.age > 30
WITH p
MATCH (p)-[:FRIEND]->(f:Person)
WHERE f.age > 30
RETURN p, f

-- 优化后：一次模式匹配
MATCH (p:Person)-[:FRIEND]->(f:Person)
WHERE p.age > 30 AND f.age > 30
RETURN p, f

-- 使用WITH减少中间结果
MATCH (p:Person)-[:FRIEND]->(friend)
WITH p, collect(friend) as friends
WHERE size(friends) > 10
RETURN p.name, size(friends) as friendCount

6.1.3 批量操作优化 ​

cypher
-- 批量创建(使用UNWIND)
UNWIND range(1, 1000) as id
CREATE (:Person {id: id, name: "Person_" + id})

-- 批量更新
MATCH (p:Person)
WHERE p.department = "OldDept"
CALL {
  WITH p
  SET p.department = "NewDept"
} IN TRANSACTIONS OF 1000 ROWS

-- 批量删除
MATCH (p:Person)
WHERE p.lastLogin < datetime().subtract(Duration.ofDays(365))
CALL {
  WITH p
  DETACH DELETE p
} IN TRANSACTIONS OF 500 ROWS

6.2 避免常见陷阱 ​

6.2.1 笛卡尔积问题 ​

cypher
-- 错误：可能导致笛卡尔积
MATCH (a:Person), (b:Person)  -- 危险！可能生成n²个结果
RETURN a.name, b.name

-- 正确：添加关系约束
MATCH (a:Person)-[:KNOWS]-(b:Person)
RETURN a.name, b.name

-- 或者明确指定条件
MATCH (a:Person), (b:Person)
WHERE a.department = b.department
RETURN a.name, b.name, a.department

6.2.2 爆炸性路径查询 ​

cypher
-- 危险：未限制的路径长度
MATCH (a:Person)-[:KNOWS*]-(b:Person)  -- 可能导致大量路径

-- 安全：限制路径长度
MATCH (a:Person)-[:KNOWS*1..3]-(b:Person)
RETURN a.name, b.name

-- 添加其他限制
MATCH path = (a:Person)-[:KNOWS*1..5]-(b:Person)
WHERE length(path) <= 5 AND a <> b
RETURN a.name, b.name, length(path)

八、实战案例 ​

8.1 社交网络分析 ​

cypher
-- 查找影响力最大的用户
MATCH (user:User)
OPTIONAL MATCH (user)-[:FOLLOWED_BY]->(follower:User)
OPTIONAL MATCH (user)-[:POSTED]->(post:Post)<-[:LIKED]-(liker:User)
WITH user,
     count(DISTINCT follower) as followers,
     count(DISTINCT liker) as likesReceived,
     count(DISTINCT post) as posts
RETURN user.username,
       followers,
       likesReceived,
       posts,
       followers * 0.6 + likesReceived * 0.3 + posts * 0.1 as influenceScore
ORDER BY influenceScore DESC
LIMIT 10

-- 发现社区
MATCH (user:User)-[:FOLLOWED_BY]->(follower:User)
CALL apoc.algo.community(20, [user], "FOLLOWED_BY", "BOTH")
YIELD node, community
WITH community, collect(node.username) as members
WHERE size(members) > 5
RETURN community, members, size(members) as communitySize
ORDER BY communitySize DESC

8.2 推荐系统 ​

cypher
-- 基于内容的推荐
MATCH (targetUser:User {id: $userId})-[:LIKED|VIEWED]->(content:Content)
MATCH (content)-[:HAS_TAG]->(tag:Tag)
WITH targetUser, tag, count(*) as score
ORDER BY score DESC
LIMIT 10
MATCH (tag)<-[:HAS_TAG]-(recommended:Content)
WHERE NOT (targetUser)-[:LIKED|VIEWED|DISLIKED]->(recommended)
WITH recommended, sum(score) as recommendationScore
RETURN recommended.title, recommendationScore
ORDER BY recommendationScore DESC
LIMIT 20

-- 协同过滤
MATCH (targetUser:User {id: $userId})-[r1:RATED]->(movie:Movie)<-[r2:RATED]-(similarUser:User)
WHERE abs(r1.rating - r2.rating) <= 1
WITH targetUser, similarUser, count(*) as commonMovies
WHERE commonMovies > 5
MATCH (similarUser)-[r3:RATED]->(recommended:Movie)
WHERE NOT (targetUser)-[:RATED]->(recommended)
WITH recommended, 
     avg(r3.rating) as avgRating,
     count(similarUser) as similarUsers
WHERE similarUsers >= 3
RETURN recommended.title, avgRating, similarUsers
ORDER BY avgRating DESC, similarUsers DESC
LIMIT 10

8.3 欺诈检测 ​

cypher
-- 检测环形交易
MATCH path = (a1:Account)-[:SENT]->(t1:Transaction)-[:RECEIVED_BY]->(a2:Account)
              -[:SENT]->(t2:Transaction)-[:RECEIVED_BY]->(a3:Account)
              -[:SENT]->(t3:Transaction)-[:RECEIVED_BY]->(a1)
WHERE t1.amount > 10000 AND t2.amount > 10000 AND t3.amount > 10000
  AND t1.timestamp < t2.timestamp < t3.timestamp
  AND duration.between(t1.timestamp, t3.timestamp).hours <= 24
RETURN a1.accountNumber as account1,
       a2.accountNumber as account2,
       a3.accountNumber as account3,
       t1.amount as amount1,
       t2.amount as amount2,
       t3.amount as amount3

-- 检测快速转账网络
MATCH (source:Account)-[t:TRANSFER]->(target:Account)
WHERE t.amount > 5000
  AND t.timestamp > datetime().subtract(Duration.ofHours(1))
WITH source, target, t.amount as amount
ORDER BY amount DESC
LIMIT 100
MATCH path = shortestPath((source)-[:OWNED_BY|CONTROLS*..3]-(target))
WHERE length(path) > 0
RETURN source.accountNumber, 
       target.accountNumber, 
       amount,
       [node IN nodes(path) WHERE node:Person | node.name] as connectedPersons
