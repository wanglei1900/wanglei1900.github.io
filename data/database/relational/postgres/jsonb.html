PostgreSQL JSONB与半结构化数据处理 完整学习笔记 ​

一、半结构化数据概述 ​

1.1 数据结构分类 ​

  • 结构化数据：固定模式，如传统关系表
  • 半结构化数据：部分结构，如JSON、XML
  • 非结构化数据：无固定结构，如文本、图像

1.2 半结构化数据特点 ​

  • 模式灵活，可动态扩展
  • 嵌套层次结构
  • 自描述性
  • 示例：日志数据、配置文件、API响应

二、PostgreSQL JSON/JSONB类型对比 ​

2.1 JSON类型 ​

sql
-- JSON类型（文本存储，保留空格、键顺序）
CREATE TABLE products_json (
    id SERIAL PRIMARY KEY,
    data JSON
);

2.2 JSONB类型（Binary JSON） ​

sql
-- JSONB类型（二进制存储，优化查询）
CREATE TABLE products_jsonb (
    id SERIAL PRIMARY KEY,
    data JSONB
);

2.3 核心差异 ​

特性JSONJSONB
存储格式文本二进制
解析速度插入快，查询慢插入慢，查询快
重复键处理保留最后一个保留最后一个
键顺序保留不保留
空格保留去除
索引支持有限完整支持
推荐用途简单存储复杂查询

三、JSONB基本操作 ​

3.1 数据插入 ​

sql
-- 直接插入JSON字符串
INSERT INTO products_jsonb (data) VALUES 
('{
  "name": "Laptop",
  "price": 999.99,
  "specs": {
    "cpu": "i7",
    "ram": "16GB",
    "storage": "512GB SSD"
  },
  "tags": ["electronics", "computer", "portable"],
  "in_stock": true,
  "dimensions": {
    "weight": 2.5,
    "height": 18,
    "width": 30,
    "depth": 2.5
  }
}');

-- 使用JSONB构建函数
INSERT INTO products_jsonb (data) VALUES 
(jsonb_build_object(
    'name', 'Smartphone',
    'price', 699.99,
    'tags', jsonb_build_array('mobile', 'electronics'),
    'metadata', jsonb_build_object('created_at', now()::text)
));

3.2 数据查询 ​

3.2.1 基本提取操作符 ​

sql
-- 获取字段（返回JSONB）
SELECT data->'name' AS name_jsonb FROM products_jsonb;
SELECT data->>'name' AS name_text FROM products_jsonb;

-- 获取嵌套字段
SELECT data->'specs'->>'cpu' AS cpu FROM products_jsonb;

-- 获取数组元素
SELECT data->'tags'->0 AS first_tag FROM products_jsonb;
SELECT data->'tags'->>0 AS first_tag_text FROM products_jsonb;

-- 路径查询
SELECT data#>>'{specs,cpu}' AS cpu_path FROM products_jsonb;

3.2.2 条件查询 ​

sql
-- 相等判断
SELECT * FROM products_jsonb WHERE data->>'name' = 'Laptop';

-- 包含判断
SELECT * FROM products_jsonb WHERE data @> '{"specs": {"cpu": "i7"}}';

-- 存在判断
SELECT * FROM products_jsonb WHERE data ? 'tags';
SELECT * FROM products_jsonb WHERE data->'tags' ? 'electronics';

-- 任意键存在
SELECT * FROM products_jsonb WHERE data ?| array['tags', 'specs'];

-- 所有键存在
SELECT * FROM products_jsonb WHERE data ?& array['name', 'price'];

3.2.3 范围查询 ​

sql
-- 价格大于500
SELECT * FROM products_jsonb 
WHERE (data->>'price')::numeric > 500;

-- 使用JSONB路径
SELECT * FROM products_jsonb 
WHERE CAST(data#>>'{dimensions,weight}' AS float) > 2.0;

3.3 数据修改 ​

3.3.1 更新操作 ​

sql
-- 合并操作（浅合并）
UPDATE products_jsonb 
SET data = data || '{"discount": 0.1}' 
WHERE id = 1;

-- 深度合并（递归合并）
UPDATE products_jsonb 
SET data = data || jsonb_build_object('specs', 
    jsonb_build_object('battery', '5000mAh')) 
WHERE id = 1;

-- 删除键
UPDATE products_jsonb 
SET data = data - 'discount' 
WHERE id = 1;

-- 删除数组元素
UPDATE products_jsonb 
SET data = data #- '{tags,0}' 
WHERE id = 1;

-- 设置特定路径值
UPDATE products_jsonb 
SET data = jsonb_set(
    data, 
    '{specs,battery}', 
    '"6000mAh"',
    true  -- 如果路径不存在则创建
) WHERE id = 1;

3.3.2 插入和追加 ​

sql
-- 插入数组元素
UPDATE products_jsonb 
SET data = jsonb_insert(
    data,
    '{tags,0}',  -- 在位置0插入
    '"new_tag"'
);

-- 追加到数组
UPDATE products_jsonb 
SET data = jsonb_set(
    data,
    '{tags}',
    (data->'tags') || '"new_tag"'::jsonb
);

四、JSONB索引优化 ​

4.1 GIN索引（通用倒排索引） ​

sql
-- 创建GIN索引（默认ops）
CREATE INDEX idx_gin_data ON products_jsonb USING gin(data);

-- 支持的操作符：@>, ?, ?|, ?&

-- 指定操作符类
CREATE INDEX idx_gin_data_path_ops ON products_jsonb 
USING gin(data jsonb_path_ops);  -- 仅支持@>

CREATE INDEX idx_gin_data_full ON products_jsonb 
USING gin(data jsonb_ops);  -- 支持所有操作符

4.2 B树索引 ​

sql
-- 在特定字段上创建B树索引
CREATE INDEX idx_btree_price ON products_jsonb 
USING btree((data->>'price')::numeric);

-- 表达式索引
CREATE INDEX idx_btree_name ON products_jsonb 
USING btree(lower(data->>'name'));

4.3 复合索引 ​

sql
-- JSONB字段与传统字段组合索引
CREATE INDEX idx_composite ON products_jsonb 
USING btree(category, (data->>'price')::numeric);

-- 多JSONB路径索引
CREATE INDEX idx_multi_jsonb ON products_jsonb 
USING btree(
    (data->>'category'),
    (data#>>'{specs,brand}')
);

4.4 索引选择策略 ​

查询类型推荐索引说明
存在性查询(?)GIN(jsonb_ops)检查键是否存在
包含查询(@>)GIN(jsonb_path_ops)效率更高
值相等查询B树表达式索引特定路径的值比较
全文搜索GIN + tsvector文本搜索
复合查询复合索引结合JSONB和传统字段

五、JSONB函数和操作符 ​

5.1 转换函数 ​

sql
-- JSON/JSONB转换
SELECT to_jsonb('{"a": 1}'::json);
SELECT jsonb_each('{"a": 1, "b": 2}'::jsonb);
SELECT jsonb_each_text('{"a": 1, "b": 2}'::jsonb);

-- 行转JSONB
SELECT to_jsonb(products) FROM products_jsonb LIMIT 1;

-- 数组处理
SELECT jsonb_array_elements('[1, 2, 3]'::jsonb);
SELECT jsonb_array_length('[1, 2, 3]'::jsonb);

5.2 聚合函数 ​

sql
-- JSONB聚合
SELECT jsonb_agg(data) FROM products_jsonb;
SELECT jsonb_object_agg(key, value) FROM (
    SELECT data->>'name' as key, data->'price' as value 
    FROM products_jsonb
) t;

-- 分组聚合
SELECT 
    data->>'category' as category,
    jsonb_agg(data->'name') as product_names
FROM products_jsonb
GROUP BY data->>'category';

5.3 验证和检查 ​

sql
-- 验证JSON有效性
SELECT jsonb_valid('{"a": 1}');  -- true
SELECT jsonb_valid('{invalid}'); -- false

-- 类型检查
SELECT jsonb_typeof('null'::jsonb);      -- 'null'
SELECT jsonb_typeof('1'::jsonb);         -- 'number'
SELECT jsonb_typeof('"text"'::jsonb);    -- 'string'
SELECT jsonb_typeof('true'::jsonb);      -- 'boolean'
SELECT jsonb_typeof('[]'::jsonb);        -- 'array'
SELECT jsonb_typeof('{}'::jsonb);        -- 'object'

5.4 文本搜索 ​

sql
-- 将JSONB转换为tsvector
ALTER TABLE products_jsonb 
ADD COLUMN data_tsvector tsvector 
GENERATED ALWAYS AS (
    to_tsvector('english', 
        coalesce(data->>'name', '') || ' ' ||
        coalesce(data->>'description', '') || ' ' ||
        coalesce(data#>>'{specs,info}', '')
    )
) STORED;

-- 创建GIN索引
CREATE INDEX idx_gin_tsvector ON products_jsonb USING gin(data_tsvector);

-- 全文搜索查询
SELECT * FROM products_jsonb 
WHERE data_tsvector @@ to_tsquery('english', 'laptop & fast');

六、JSONB模式验证 ​

6.1 CHECK约束 ​

sql
-- 基础验证
ALTER TABLE products_jsonb 
ADD CONSTRAINT valid_json_structure 
CHECK (
    data ? 'name' AND 
    data ? 'price' AND
    (data->>'price')::numeric > 0 AND
    jsonb_typeof(data->'tags') = 'array'
);

-- 使用JSON Schema验证
CREATE EXTENSION IF NOT EXISTS jsquery;

ALTER TABLE products_jsonb 
ADD CONSTRAINT valid_json_schema 
CHECK (
    data @? '$.name ? (@ != null)' AND
    data @? '$.price ? (@ > 0)' AND
    data @? '$.tags ? (@.type() == "array")'
);

6.2 触发器验证 ​

sql
CREATE OR REPLACE FUNCTION validate_product_jsonb()
RETURNS TRIGGER AS $$
BEGIN
    -- 验证必需字段
    IF NOT (NEW.data ? 'name') THEN
        RAISE EXCEPTION 'Missing required field: name';
    END IF;
    
    -- 验证价格格式
    IF NOT (NEW.data @? '$.price ? (@ >= 0)') THEN
        RAISE EXCEPTION 'Price must be non-negative';
    END IF;
    
    -- 验证标签是数组
    IF NOT (jsonb_typeof(NEW.data->'tags') = 'array') THEN
        RAISE EXCEPTION 'Tags must be an array';
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tr_validate_jsonb
BEFORE INSERT OR UPDATE ON products_jsonb
FOR EACH ROW EXECUTE FUNCTION validate_product_jsonb();

七、高级查询技巧 ​

7.1 连接JSONB表 ​

sql
-- JSONB字段与传统表连接
SELECT 
    p.data->>'name' as product_name,
    c.name as category_name
FROM products_jsonb p
JOIN categories c ON p.data->>'category_id' = c.id::text;

-- 自连接JSONB数组
SELECT 
    o1.data->>'order_id' as order_id,
    jsonb_agg(p.data->>'name') as products
FROM orders_jsonb o1
CROSS JOIN LATERAL jsonb_array_elements(o1.data->'product_ids') AS pid
JOIN products_jsonb p ON p.data->>'id' = pid::text
GROUP BY o1.data->>'order_id';

7.2 递归查询JSONB ​

sql
-- 递归展开嵌套结构
WITH RECURSIVE flatten_tree AS (
    SELECT 
        id,
        data->>'name' as name,
        data->>'parent_id' as parent_id,
        1 as level
    FROM categories_jsonb
    WHERE data->>'parent_id' IS NULL
    
    UNION ALL
    
    SELECT 
        c.id,
        c.data->>'name',
        c.data->>'parent_id',
        ft.level + 1
    FROM categories_jsonb c
    JOIN flatten_tree ft ON c.data->>'parent_id' = ft.data->>'id'
)
SELECT * FROM flatten_tree ORDER BY level, name;

7.3 窗口函数与JSONB ​

sql
-- JSONB数据窗口分析
SELECT 
    data->>'category' as category,
    data->>'name' as product_name,
    (data->>'price')::numeric as price,
    AVG((data->>'price')::numeric) OVER (
        PARTITION BY data->>'category'
    ) as avg_category_price,
    RANK() OVER (
        PARTITION BY data->>'category' 
        ORDER BY (data->>'price')::numeric DESC
    ) as price_rank
FROM products_jsonb;

八、性能优化策略 ​

8.1 查询优化技巧 ​

sql
-- 1. 使用投影减少数据传输
EXPLAIN ANALYZE
SELECT data->>'name', data->>'price' 
FROM products_jsonb 
WHERE data @> '{"category": "electronics"}';

-- 2. 避免在WHERE子句中转换类型
-- 不佳：类型转换阻止索引使用
SELECT * FROM products_jsonb 
WHERE (data->>'price')::numeric > 100;

-- 推荐：使用JSONB操作符
SELECT * FROM products_jsonb 
WHERE data @> '{"price": 100}';

-- 3. 分页优化
WITH indexed_data AS (
    SELECT 
        id,
        data,
        row_number() OVER (ORDER BY id) as rn
    FROM products_jsonb
    WHERE data @> '{"in_stock": true}'
)
SELECT * FROM indexed_data WHERE rn BETWEEN 1 AND 20;

九、实际应用场景 ​

9.1 电商产品目录 ​

sql
-- 灵活的产品属性存储
CREATE TABLE products_catalog (
    id SERIAL PRIMARY KEY,
    sku VARCHAR(50) UNIQUE,
    attributes JSONB,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- 创建多列索引
CREATE INDEX idx_attrs_search ON products_catalog 
USING gin(attributes jsonb_path_ops);

-- 支持动态过滤
CREATE OR REPLACE FUNCTION search_products(
    p_category TEXT DEFAULT NULL,
    p_min_price NUMERIC DEFAULT NULL,
    p_features JSONB DEFAULT NULL
) RETURNS SETOF products_catalog AS $$
BEGIN
    RETURN QUERY
    SELECT * FROM products_catalog pc
    WHERE (p_category IS NULL OR pc.attributes->>'category' = p_category)
      AND (p_min_price IS NULL OR 
           (pc.attributes->>'price')::numeric >= p_min_price)
      AND (p_features IS NULL OR pc.attributes @> p_features)
    ORDER BY (pc.attributes->>'popularity')::numeric DESC;
END;
$$ LANGUAGE plpgsql;

9.2 日志和事件存储 ​

-- 结构化的日志存储
CREATE TABLE event_logs (
    id BIGSERIAL PRIMARY KEY,
    event_time TIMESTAMPTZ DEFAULT NOW(),
    source VARCHAR(100),
    level VARCHAR(20),
    metadata JSONB,
    message TEXT
) PARTITION BY RANGE (event_time);

-- 创建分区
CREATE TABLE event_logs_2024_01 PARTITION OF event_logs
FOR VALUES FROM ('2024-01-01') TO ('2024-02-01');

-- 创建复合索引
CREATE INDEX idx_logs_search ON event_logs 
USING btree(event_time, level) 
WHERE level IN ('ERROR', 'WARNING');

CREATE INDEX idx_logs_metadata ON event_logs 
USING gin(metadata);

-- 日志分析查询
SELECT 
    date_trunc('hour', event_time) as hour,
    level,
    COUNT(*) as count,
    jsonb_object_agg(
        metadata->>'error_code',
        COUNT(*)
    ) as errors_by_code
FROM event_logs
WHERE event_time >= NOW() - INTERVAL '24 hours'
GROUP BY 1, 2
ORDER BY 1 DESC, 3 DESC;

9.3 配置管理 ​

sql
-- 多租户配置存储
CREATE TABLE tenant_configurations (
    tenant_id VARCHAR(50),
    config_type VARCHAR(50),
    config_data JSONB,
    version INTEGER DEFAULT 1,
    is_active BOOLEAN DEFAULT true,
    effective_from TIMESTAMPTZ DEFAULT NOW(),
    effective_to TIMESTAMPTZ,
    PRIMARY KEY (tenant_id, config_type, version)
);

-- 获取当前有效配置
CREATE OR REPLACE FUNCTION get_tenant_config(
    p_tenant_id VARCHAR,
    p_config_type VARCHAR
) RETURNS JSONB AS $$
DECLARE
    v_config JSONB;
BEGIN
    SELECT config_data INTO v_config
    FROM tenant_configurations
    WHERE tenant_id = p_tenant_id
      AND config_type = p_config_type
      AND is_active = true
      AND (effective_to IS NULL OR effective_to > NOW())
    ORDER BY version DESC
    LIMIT 1;
    
    RETURN v_config;
END;
$$ LANGUAGE plpgsql STABLE;

十、最佳实践与注意事项 ​

10.1 设计建议 ​

  1. 混合使用：结合传统表和JSONB，关键数据用列存储，灵活属性用JSONB
  2. 模式设计：定义JSONB字段的基本结构约定
  3. 版本控制：为JSONB结构变化设计版本字段
  4. 验证机制：实施数据验证确保数据质量

10.2 性能注意事项 ​

  1. 索引策略：根据查询模式选择合适的索引类型
  2. 避免过度嵌套：深层次嵌套影响查询性能
  3. 定期维护：对JSONB表进行VACUUM和REINDEX
  4. 监控大小：JSONB文档不宜过大（建议<1MB）

10.3 迁移策略 ​

总结 ​

PostgreSQL的JSONB类型为半结构化数据处理提供了强大的支持。通过合理使用JSONB操作符、索引策略和优化技巧，可以在保持关系数据库优势的同时，享受NoSQL的灵活性。关键是根据具体应用场景选择合适的数据建模方式，混合使用传统关系模型和JSONB，实现最佳的性能和灵活性平衡。

记住：JSONB不是万能的，它最适合存储可选的、可变的、或稀疏的属性，而核心业务数据仍应使用传统的规范化表结构。
